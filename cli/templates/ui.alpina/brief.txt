type Blip = { f: number; to?: number; d?: number; g?: number; type?: OscillatorType; at?: number };
function useBlip(on: boolean) {
  const ctx = useRef<AudioContext | null>(null);
  useEffect(() => () => void ctx.current?.close().catch(() => {}), []);
  return useCallback(
    (b: Blip) => {
      if (!on || typeof window === "undefined") return;
      if (document.documentElement.dataset.cdMute === "1") return;
      const ua = (navigator as unknown as { userActivation?: { hasBeenActive: boolean } }).userActivation;
      if (ua && !ua.hasBeenActive) return;

        const AC = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
        if (!AC) return;
        const a = (ctx.current ??= new AC());
        if (a.state === "suspended") void a.resume();
        const t = a.currentTime + (b.at ?? 0);
        const d = Math.min(b.d ?? 0.04, 0.06);
        const amp = a.createGain();
        amp.gain.setValueAtTime(0.0001, t);
        amp.gain.exponentialRampToValueAtTime(Math.min(b.g ?? 0.035, 0.06), t + 0.005);
        amp.gain.exponentialRampToValueAtTime(0.0001, t + d);
        amp.connect(a.destination);
        const o = a.createOscillator();
        o.type = b.type ?? "sine";
        o.frequency.setValueAtTime(b.f, t);
        if (b.to) o.frequency.exponentialRampToValueAtTime(b.to, t + d);
        o.connect(amp);
        o.start(t);
        o.stop(t + d + 0.01);
      } catch {
        /* no sound is fine */
      }
    },
    [on],
  );
}

/* ---------- Geometry ---------- */
type Pt = { x: number; y: number; m: number; h: number; g: number };

/* Densify the track (Catmull-Rom) and measure it in metres. */
function buildTrack() {
  const out: Pt[] = [];
  const n = TRACK.length;
  const P = (i: number) => TRACK[Math.max(0, Math.min(n - 1, i))];
  for (let i = 0; i < n - 1; i++) {
    const [p0, p1, p2, p3] = [P(i - 1), P(i), P(i + 1), P(i + 2)];
    const steps = 10;
    for (let s = 0; s < steps; s++) {
      const t = s / steps;
      const t2 = t * t;
      const t3 = t2 * t;
      const cr = (a: number, b: number, c: number, d: number) =>
        0.5 * (2 * b + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t2 + (-a + 3 * b - 3 * c + d) * t3);
      out.push({ x: cr(p0[0], p1[0], p2[0], p3[0]), y: cr(p0[1], p1[1], p2[1], p3[1]), m: 0, h: p1[3] + (p2[3] - p1[3]) * t, g: p1[2] + (p2[2] - p1[2]) * t });
    }
  }
  const last = TRACK[n - 1];
  out.push({ x: last[0], y: last[1], m: 0, h: last[3], g: last[2] });
  for (let i = 1; i < out.length; i++) {
    const a = out[i - 1];
    const b = out[i];
    const ds = Math.hypot(b.x - a.x, b.y - a.y);
    b.m = a.m + ds / ((a.g + b.g) / 2);
  }
  return out;
}

/* A point on the track at metre m (clamped to the ends). */
function at(tr: Pt[], m: number): Pt {
  if (m <= 0) return tr[0];
  const L = tr[tr.length - 1].m;
  if (m >= L) return tr[tr.length - 1];
  let lo = 0;
  let hi = tr.length - 1;
  while (hi - lo > 1) {
    const mid = (lo + hi) >> 1;
    if (tr[mid].m < m) lo = mid;
    else hi = mid;
  }
  const a = tr[lo];
  const b = tr[hi];
  const t = (m - a.m) / (b.m - a.m || 1);
  return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t, m, h: a.h + (b.h - a.h) * t, g: a.g + (b.g - a.g) * t };
}

/* The line under the sprite as it sat in the photo, at plate x. */
function srcAt(x: number) {
  let i = 0;
  while (i < SRC.length - 2 && x > SRC[i + 1][0]) i++;
  const a = SRC[i];
  const b = SRC[i + 1];
  const t = (x - a[0]) / (b[0] - a[0]);
  return { y: a[1] + (b[1] - a[1]) * t, g: a[2] + (b[2] - a[2]) * t, h: a[3] + (b[3] - a[3]) * t };
}

/* Each car, cut into thin strips so it follows the curve, with each strip's
   length in metres (measured where it sat in the photo). */
type Strip = { sx0: number; sx1: number; m0: number; m1: number; hs: number; a: [number, number]; b: [number, number] };
type Car = { strips: Strip[]; len: number };
function buildCars(): Car[] {
  return CARS.map(([c0, c1]) => {
    const N = 8;
    const strips: Strip[] = [];
    let m = 0;
    for (let i = 0; i < N; i++) {
      const sx0 = c0 + ((c1 - c0) * i) / N;
      const sx1 = c0 + ((c1 - c0) * (i + 1)) / N;
      const x0 = SPR.x0 + sx0 / SPR.s;
      const x1 = SPR.x0 + sx1 / SPR.s;
      const A = srcAt(x0);
      const B = srcAt(x1);
      const len = Math.hypot(x1 - x0, B.y - A.y) / ((A.g + B.g) / 2);
      strips.push({ sx0, sx1, m0: m, m1: m + len, hs: (A.h + B.h) / 2, a: [x0, A.y], b: [x1, B.y] });
      m += len;
    }
    return { strips, len: m };
  });
}

/* Plate → screen: cover with a focus point and a little overscan for parallax. */
type PMap = { s: number; ox: number; oy: number };
function coverMap(W: number, H: number): PMap {
  const phone = W < 640;
  const fx = phone ? 0.68 : 0.5;
  const fy = phone ? 0.52 : 0.5;
  const s = Math.max(W / PW, H / PH) * (phone ? 1.08 : 1.06);
  const iw = PW * s;
  const ih = PH * s;
  const ox = Math.min(0, Math.max(W - iw, W / 2 - fx * iw));
  const oy = Math.min(0, Math.max(H - ih, H / 2 - fy * ih));
  return { s, ox, oy };
}

function loadImg(src: string) {
  return new Promise<HTMLImageElement>((res, rej) => {
    const i = new Image();
    i.crossOrigin = "anonymous";
    i.decoding = "async";
    i.onload = () => res(i);
    i.onerror = rej;
    i.src = src;
  });
}

/* A 256² tiling value-noise texture: four independent channels, 16 smooth
   cells across, so magnified octaves stay soft (no facets, no contour lines). */
function noiseData(size: number, seed: number) {
  let s = seed >>> 0;
  const rnd = () => {
    s = (s * 1664525 + 1013904223) >>> 0;
    return s / 4294967296;
  };
  const out = new Uint8Array(size * size * 4);
  const g = 16;
  for (let c = 0; c < 4; c++) {
    const lat = Array.from({ length: g * g }, rnd);
    const v = (xx: number, yy: number) => lat[(((yy % g) + g) % g) * g + (((xx % g) + g) % g)];
    for (let y = 0; y < size; y++)
      for (let x = 0; x < size; x++) {
        const fx = (x / size) * g;
        const fy = (y / size) * g;
        const ix = Math.floor(fx);
        const iy = Math.floor(fy);
        const tx = fx - ix;
        const ty = fy - iy;
        const ux = tx * tx * tx * (tx * (tx * 6 - 15) + 10);
        const uy = ty * ty * ty * (ty * (ty * 6 - 15) + 10);
        const a = v(ix, iy) + (v(ix + 1, iy) - v(ix, iy)) * ux;
        const b = v(ix, iy + 1) + (v(ix + 1, iy + 1) - v(ix, iy + 1)) * ux;
        out[(y * size + x) * 4 + c] = Math.round((a + (b - a) * uy) * 255);
      }
  }
  return out;
}

/* The plate in two depths, mist rolling through the valley, cloud drifting
   past the peaks, the low sun breathing. Eight texture reads at most. */
const FRAG = `
precision highp float;
varying vec2 vUv;
uniform sampler2D uPlate, uMask, uNoise;
uniform vec2 uRes;      // canvas css px
uniform vec3 uMap;      // ox, oy, scale (css px)
uniform vec2 uFar, uMid; // parallax offsets, css px
uniform float uT;
float fbm(vec2 p){
  return texture2D(uNoise, p).r * .5
       + texture2D(uNoise, p * 2.03 + vec2(.37, .11)).g * .27
       + texture2D(uNoise, p * 4.11 + vec2(.71, .53)).b * .15
       + texture2D(uNoise, p * 8.27 + vec2(.13, .89)).a * .08;
}
void main(){
  vec2 px = vec2(vUv.x, 1.0 - vUv.y) * uRes;
  vec2 size = vec2(${PW}.0, ${PH}.0) * uMap.z;
  vec2 uv0 = (px - uMap.xy) / size;
  float d = texture2D(uMask, uv0).r;
  vec2 uv = uv0 - mix(uFar, uMid, d) / size;
  vec3 col = texture2D(uPlate, uv).rgb;
  vec4 mk = texture2D(uMask, uv);
  // mist: soft banks rolling along the valley, folding over themselves
  // (the noise texture has 16 cells per repeat: q is in repeats)
  if (mk.g > .01) {
    vec2 q = vec2(uv.x * .32 - uT * .0034, uv.y * 1.05);
    float w = texture2D(uNoise, q * .45 + vec2(uT * .0012, -uT * .0008)).r;
    float n = fbm(q + vec2(w * .045, w * .016));
    float a = mk.g * smoothstep(.38, .68, n);
    float lit = clamp(1.05 - uv.x * .9, 0.0, 1.0);
    vec3 mc = mix(vec3(.82, .85, .89), vec3(1.0, .92, .82), lit);
    col = mix(col, mc, a * .82);
  }
  // cloud on the peaks: wisps drifting slowly right to left
  if (mk.b > .01) {
    vec2 q = vec2(uv.x * .38 + uT * .0026, uv.y * 1.5);
    float w = texture2D(uNoise, q * .5 - vec2(uT * .0009, 0.0)).g;
    float n = fbm(q + vec2(w * .04, w * .012));
    float a = mk.b * smoothstep(.40, .66, n);
    col = mix(col, vec3(1.0, .95, .93), a * .78);
  }
  // the low sun, breathing slowly, and shafts of its light fanning over the valley
  vec2 dir = (uv - vec2(-.02, .33)) * vec2(1.5, 1.0);
  float ang = atan(dir.y, dir.x);
  float ray = texture2D(uNoise, vec2(ang * .55 + uT * .0016, .37)).r * .65 + texture2D(uNoise, vec2(ang * 1.3 - uT * .0011, .71)).g * .35;
  float shaft = smoothstep(.48, .82, ray) * exp(-length(dir) * 2.1) * smoothstep(-.01, .08, dir.y) * (1.0 - d * .55);
  col += vec3(1.0, .76, .46) * shaft * .2;
  vec2 sp = (uv - vec2(.0, .32)) * vec2(1.1, 2.0);
  col += vec3(1.0, .70, .40) * exp(-dot(sp, sp) * 7.0) * (.09 + .035 * sin(uT * .45));
  gl_FragColor = vec4(col, 1.0);
}`;

    /* --- WebGL --- */
    const gl = cv.getContext("webgl", { antialias: false, alpha: false, premultipliedAlpha: false }) as WebGLRenderingContext | null;
    let prog: WebGLProgram | null = null;
    const U: Record<string, WebGLUniformLocation | null> = {};
    const tex: WebGLTexture[] = [];
    let glReady = false;
    if (gl) {
      const sh = (t: number, src: string) => {
        const s = gl.createShader(t)!;
        gl.shaderSource(s, src);
        gl.compileShader(s);
        return s;
      };
      prog = gl.createProgram()!;
      gl.attachShader(prog, sh(gl.VERTEX_SHADER, VERT));
      gl.attachShader(prog, sh(gl.FRAGMENT_SHADER, FRAG));
      gl.linkProgram(prog);
      if (!gl.getProgramParameter(prog, gl.LINK_STATUS)) prog = null;
      if (prog) {
        gl.useProgram(prog);
        const buf = gl.createBuffer();
        gl.bindBuffer(gl.ARRAY_BUFFER, buf);
        gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]), gl.STATIC_DRAW);
        const loc = gl.getAttribLocation(prog, "aPos");
        gl.enableVertexAttribArray(loc);
        gl.vertexAttribPointer(loc, 2, gl.FLOAT, false, 0, 0);
        for (const n of ["uPlate", "uMask", "uNoise", "uRes", "uMap", "uFar", "uMid", "uT"]) U[n] = gl.getUniformLocation(prog, n);
        gl.pixelStorei(gl.UNPACK_COLORSPACE_CONVERSION_WEBGL, gl.NONE);
        const mk = (unit: number, src: TexImageSource | null, data?: Uint8Array, size?: number) => {
          const t = gl.createTexture()!;
          gl.activeTexture(gl.TEXTURE0 + unit);
          gl.bindTexture(gl.TEXTURE_2D, t);
          if (data && size) gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, size, size, 0, gl.RGBA, gl.UNSIGNED_BYTE, data);
          else if (src) gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, src);
          const pot = !!data;
          if (pot) gl.generateMipmap(gl.TEXTURE_2D);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, pot ? gl.LINEAR_MIPMAP_LINEAR : gl.LINEAR);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, pot ? gl.REPEAT : gl.CLAMP_TO_EDGE);
          gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, pot ? gl.REPEAT : gl.CLAMP_TO_EDGE);
          tex.push(t);
        };
        mk(2, null, noiseData(256, 11), 256);
        Promise.all([loadImg(plate), loadImg(masks)])
          .then(([a, b]) => {
            if (dead || !prog) return;
            gl.useProgram(prog);
            mk(0, a);
            mk(1, b);
            gl.uniform1i(U.uPlate, 0);
            gl.uniform1i(U.uMask, 1);
            gl.uniform1i(U.uNoise, 2);
            glReady = true;
            draw(last);
          })
          .catch(() => {});
      }
    }

    /* --- The train --- */
    const tx = tc.getContext("2d");
    let spr: HTMLImageElement | null = null;
    let fgImg: HTMLImageElement | null = null;
    loadImg(train)
      .then((i) => {
        spr = i;
        draw(last);
      })
      .catch(() => {});
    loadImg(fg)
      .then((i) => {
        fgImg = i;
        draw(last);
      })
      .catch(() => {});

    const drawTrain = (headM: number, mx: number, my: number) => {
      const img = spr;
      if (!tx || !img) return;
      const s = map.s;
      const kx = img.naturalWidth / SPR.w;
      const ky = img.naturalHeight / SPR.h;
      let m = headM;
      CONSIST.forEach((ci, idx) => {
        const car = cars[ci];
        if (m > L + 2 || m + car.len < -2) {
          m += car.len;
          return;
        }
        for (const st of car.strips) {
          const A = at(track, m + st.m0);
          const B = at(track, m + st.m1);
          if (st.m1 + m < 0 || st.m0 + m > L) continue;
          const ht = (A.h + B.h) / 2;
          // source frame (u_s, v_s) → target frame (u_t, v_t)
          const usx = st.b[0] - st.a[0];
          const usy = st.b[1] - st.a[1];
          const vsy = -st.hs;
          const utx = B.x - A.x;
          const uty = B.y - A.y;
          const vty = -ht;
          // inverse of [[usx, 0],[usy, vsy]]
          const det = usx * vsy;
          if (Math.abs(det) < 1e-6) continue;
          const i00 = vsy / det;
          const i10 = -usy / det;
          const i11 = usx / det;
          // M = [[utx, 0],[uty, vty]] · inv
          const a = utx * i00;
          const b = uty * i00 + vty * i10;
          const c = 0;
          const d = vty * i11;
          const e = A.x - (a * st.a[0] + c * st.a[1]);
          const f = A.y - (b * st.a[0] + d * st.a[1]);
          tx.setTransform(dpr * s * a, dpr * s * b, dpr * s * c, dpr * s * d, dpr * (s * e + map.ox + mx), dpr * (s * f + map.oy + my));
          const ov = 0.9; // overlap, sprite px, so strips never seam
          const sx0 = Math.max(0, st.sx0 - ov);
          const sx1 = Math.min(SPR.w, st.sx1 + ov);
          tx.drawImage(img, sx0 * kx, 0, (sx1 - sx0) * kx, SPR.h * ky, SPR.x0 + sx0 / SPR.s, SPR.y0, (sx1 - sx0) / SPR.s, SPR.h / SPR.s);
        }
        // headlamps on the locomotive
        if (idx === 0) {
          const st = car.strips[0];
          const A = at(track, m + st.m0);
          const B = at(track, m + st.m1);
          const ht = (A.h + B.h) / 2;
          const k = ht / st.hs;
          for (const [lx, ly] of LAMPS) {
            const px = SPR.x0 + lx / SPR.s;
            const py = SPR.y0 + ly / SPR.s;
            const fx = (px - st.a[0]) / (st.b[0] - st.a[0]);
            const up = srcAt(px).y - py;
            const X = A.x + (B.x - A.x) * fx;
            const Y = A.y + (B.y - A.y) * fx - up * k;
            const sx = (s * X + map.ox + mx) * dpr;
            const sy = (s * Y + map.oy + my) * dpr;
            const r = 7 * k * s * dpr;
            tx.setTransform(1, 0, 0, 1, 0, 0);
            const g = tx.createRadialGradient(sx, sy, 0, sx, sy, r);
            g.addColorStop(0, "rgba(255,248,226,.95)");
            g.addColorStop(0.22, "rgba(255,226,170,.45)");
            g.addColorStop(1, "rgba(255,200,120,0)");
            tx.globalCompositeOperation = "lighter";
            tx.fillStyle = g;
            tx.fillRect(sx - r, sy - r, r * 2, r * 2);
            tx.globalCompositeOperation = "source-over";
          }
        }
        m += car.len;
      });
    };

    /* --- One frame --- */
    let last = reduced ? 3 : 0;
    const par = { x: 0, y: 0 };
    const draw = (t: number) => {
      const s = map.s;
      const sc = Math.min(1, W / 1440) * 0.6 + 0.4;
      const far = { x: -par.x * 3 * sc, y: -par.y * 2 * sc };
      const mid = { x: -par.x * 12 * sc, y: -par.y * 6 * sc };
      const nr = { x: -par.x * 30 * sc, y: -par.y * 12 * sc };
      if (gl && prog && glReady) {
        if (!still.dataset.hide) {
          still.dataset.hide = "1";
          still.style.opacity = "0";
        }
        gl.viewport(0, 0, cv.width, cv.height);
        gl.uniform2f(U.uRes, W, H);
        gl.uniform3f(U.uMap, map.ox, map.oy, s);
        gl.uniform2f(U.uFar, far.x, far.y);
        gl.uniform2f(U.uMid, mid.x, mid.y);
        gl.uniform1f(U.uT, t);
        gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
      } else {
        still.style.transform = `translate3d(${mid.x}px, ${mid.y}px, 0)`;
      }
      if (tx) {
        tx.setTransform(1, 0, 0, 1, 0, 0);
        tx.clearRect(0, 0, tc.width, tc.height);
        const v = BASE_SPEED * speedRef.current;
        // the train sits where the photo has it 3 s after it appears
        const head0 = noseM - v * (t - 3);
        let h = ((head0 % spacing) + spacing) % spacing;
        h -= spacing;
        for (; h < L + 2; h += spacing) if (h + trainLen > -2) drawTrain(h, mid.x, mid.y);
        // trees and the pole that stand in front of the track, at viaduct depth
        if (fgImg) {
          tx.setTransform(dpr, 0, 0, dpr, 0, 0);
          const kx = fgImg.naturalWidth / FG.cols;
          tx.drawImage(
            fgImg,
            0,
            0,
            FG.cols * kx,
            FG.midRows * kx,
            map.ox + FG.x0 * s + mid.x,
            map.oy + FG.midY0 * s + mid.y,
            (FG.x1 - FG.x0) * s,
            (FG.midRows / FG.k) * s,
          );
        }
      }
      near.style.transform = `translate3d(${nr.x}px, ${nr.y}px, 0)`;
      type.style.transform = `translate3d(${par.x * 10 * sc}px, ${par.y * 5 * sc}px, 0)`;
    };

    let t0 = performance.now();
    let prev = t0;
    const loop = (now: number) => {
      raf = 0;
      if (dead) return;
      if (!visible || document.hidden) {
        t0 += now - prev; // freeze time while hidden
        prev = now;
        return;
      }
      const dt = Math.min(0.05, (now - prev) / 1000);
      prev = now;
      const t = (now - t0) / 1000;
      last = t;
      // a slow ambient dolly, plus the pointer
      const ax = Math.sin(t * 0.39) * 0.42 + Math.sin(t * 0.17 + 1.3) * 0.18;
      const ay = Math.sin(t * 0.27 + 0.6) * 0.3;
      const gx = ptr.on ? ptr.tx * 0.85 + ax * 0.35 : ax;
      const gy = ptr.on ? ptr.ty * 0.85 + ay * 0.35 : ay;
      const k = 1 - Math.exp(-dt * 2.6);
      par.x += (gx - par.x) * k;
      par.y += (gy - par.y) * k;
      draw(t);
      raf = requestAnimationFrame(loop);
    };

    const onMove = (e: PointerEvent) => {
      const r = el.getBoundingClientRect();
      ptr.tx = Math.max(-1, Math.min(1, ((e.clientX - r.left) / r.width) * 2 - 1));
      ptr.ty = Math.max(-1, Math.min(1, ((e.clientY - r.top) / r.height) * 2 - 1));
      ptr.on = true;
    };
    const onLeave = () => {
      ptr.on = false;
    };
    el.addEventListener("pointermove", onMove);
    el.addEventListener("pointerleave", onLeave);

    if (reduced) {
      draw(3);
    } else {
      raf = requestAnimationFrame((n) => {
        t0 = n;
        prev = n;
        loop(n);
      });
    }

    return () => {
      dead = true;
      cancelAnimationFrame(raf);
      ro.disconnect();
      io.disconnect();
      document.removeEventListener("visibilitychange", onVis);
      el.removeEventListener("pointermove", onMove);
      el.removeEventListener("pointerleave", onLeave);
      if (gl) {
        for (const t of tex) gl.deleteTexture(t);
        if (prog) gl.deleteProgram(prog);
      }
    };
  }, [plate, train, fg, masks]);

  /* --- Booking --- */
  const pick = useCallback(
    (i: number) => {
      setSel(i);
      setOpen(false);
      blip({ f: 880, to: 1320, d: 0.035, g: 0.03, type: "triangle" });
      dateBtn.current?.focus();
    },
    [blip],
  );
  useEffect(() => {
    if (!open) return;
    setFocus(sel);
    const onDown = (e: PointerEvent) => {
      const t = e.target as Node;
      if (listRef.current?.contains(t) || dateBtn.current?.contains(t)) return;
      setOpen(false);
    };
    const r = root.current;
    r?.addEventListener("pointerdown", onDown);
    return () => r?.removeEventListener("pointerdown", onDown);
  }, [open, sel]);
  useEffect(() => {
    if (open) listRef.current?.querySelector<HTMLButtonElement>(`[data-i="${focus}"]`)?.focus();
  }, [open, focus]);
  const onListKey = (e: KeyboardEvent<HTMLDivElement>) => {
    if (e.key === "ArrowRight" || e.key === "ArrowDown") {
      e.preventDefault();
      setFocus((f) => Math.min(DEPS.length - 1, f + 1));
    } else if (e.key === "ArrowLeft" || e.key === "ArrowUp") {
      e.preventDefault();
      setFocus((f) => Math.max(0, f - 1));
    } else if (e.key === "Escape") {
      e.preventDefault();
      setOpen(false);
      dateBtn.current?.focus();
    }
  };
  const book = () => {
    if (booked) return;
    setBooked(true);
    blip({ f: 659, d: 0.06, g: 0.04, type: "sine" });
    blip({ f: 988, d: 0.06, g: 0.035, type: "sine", at: 0.13 });
    onBook?.({ date: `${DEPS[sel].day} ${DEPS[sel].date}`, fare: DEPS[sel].fare });
    window.clearTimeout(bookedTimer.current);
    // Resets so the next visitor can try. Remove to keep it booked.
    bookedTimer.current = window.setTimeout(() => setBooked(false), 2200);
  };
  useEffect(() => () => window.clearTimeout(bookedTimer.current), []);

  const dep = DEPS[sel];
  const vars = { "--alp-accent": accent } as CSSProperties;

  return (
    <div ref={root} className="alp-root" style={vars}>
      <style>{css}</style>
      <div className="alp-scene" aria-hidden="true">
        <div ref={stillRef} className="alp-still" style={{ backgroundImage: `url(${plate})` }} />
        <canvas ref={glRef} className="alp-gl" />
        <canvas ref={trRef} className="alp-train" />
        <div ref={nearRef} className="alp-near" style={{ backgroundImage: `url(${fg})` }} />
        <div className="alp-shade" />
        <div className="alp-grain" />
      </div>

      <header className="alp-nav">
        <a className="alp-logo" href="#" onClick={(e) => e.preventDefault()}>
          <svg width="20" height="20" viewBox="0 0 20 20" aria-hidden="true">
            <path d="M2 16 L8 6 L11 11 L13 8 L18 16 Z" fill="none" stroke="currentColor" strokeWidth="1.7" strokeLinejoin="round" />
          </svg>
          {brand}
        </a>
        <nav className="alp-links" aria-label="Main">
          <a href="#" onClick={(e) => e.preventDefault()}>Routes</a>
          <a href="#" onClick={(e) => e.preventDefault()}>Cabins</a>
          <a href="#" onClick={(e) => e.preventDefault()}>Dining</a>
        </nav>
        <button type="button" className="alp-navbtn">{cta}</button>
      </header>

      <div ref={typeRef} className="alp-type">
        <h1 className="alp-h1">
          <span className="alp-line"><span>Sleep through</span></span>
          <span className="alp-line"><span>the Alps.</span></span>
        </h1>
        <p className="alp-sub">Zurich to Venice overnight. Board at 21:40, wake up at the lagoon.</p>
      </div>

      <div className="alp-book" role="group" aria-label="Book a sleeper">
        <span className="alp-seg alp-route">
          Zurich <span aria-hidden="true">→</span><span className="alp-sr"> to </span> Venice
        </span>
        <button
          ref={dateBtn}
          type="button"
          className="alp-seg alp-date"
          aria-haspopup="listbox"
          aria-expanded={open}
          aria-controls={`alp-list-${uid}`}
          onClick={() => {
            setOpen((o) => !o);
            blip({ f: 620, d: 0.03, g: 0.025, type: "triangle" });
          }}
        >
          {dep.day} {dep.date}
          <svg width="10" height="10" viewBox="0 0 10 10" aria-hidden="true">
            <path d="M2 4 L5 7 L8 4" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" />
          </svg>
        </button>
        <span className="alp-seg alp-fare" aria-live="polite">
          Sleeper for 2 · <span key={dep.fare} className="alp-num">€{dep.fare}</span>
        </span>
        <button type="button" className={`alp-go${booked ? " is-on" : ""}`} onClick={book}>
          <span className="alp-go-a">Book</span>
          <span className="alp-go-b" aria-hidden={!booked}>
            <svg width="13" height="13" viewBox="0 0 14 14" aria-hidden="true">
              <path d="M2.5 7.3 L5.6 10.2 L11.5 3.8" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
            </svg>
            Held
          </span>
        </button>

        <div
          id={`alp-list-${uid}`}
          ref={listRef}
          className={`alp-list${open ? " is-open" : ""}`}
          role="listbox"
          aria-label="Departures from Zurich, 21:40"
          onKeyDown={onListKey}
          hidden={!open}
        >
          <div className="alp-list-h">Departures · 21:40 from Zürich HB</div>
          <div className="alp-days">
            {DEPS.map((d, i) => (
              <button
                key={d.date}
                type="button"
                role="option"
                aria-selected={i === sel}
                data-i={i}
                tabIndex={i === focus ? 0 : -1}
                className={`alp-day${i === sel ? " is-sel" : ""}`}
                onClick={() => pick(i)}
                onKeyDown={(e) => {
                  if (e.key === "Enter" || e.key === " ") {
                    e.preventDefault();
                    pick(i);
                  }
                }}
              >
                <span className="alp-day-d">{d.day}</span>
                <span className="alp-day-n">{d.date.split(" ")[0]}</span>
                <span className="alp-day-p">€{d.fare}</span>
                <span className={`alp-day-l${d.left <= 2 ? " is-low" : ""}`}>{d.left} left</span>
              </button>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}

The full brief of this design, for reference, or to rebuild it in another stack:

# Alpina · night-train hero

The first screen of a night-train company running sleepers from Zurich to Venice through the Alps. The picture is a real place at sunrise: a curved stone viaduct high over a gorge, snow peaks lit pink behind, mist lying in the valley and a village with a white church far below. It is alive, like footage: the red train really travels across the viaduct, car by car, getting larger as it comes towards you and slipping behind the trees at both ends so the loop never shows; mist rolls through the valley, wisps of cloud drift on the peaks, shafts of low sun fan across the valley, and the scene is built in planes that part slowly on their own and more under the pointer. The one piece of product UI is the booking pill: pick a departure and the fare updates, Book holds the cabin.

## Images (one generated plate, three files cut from it)

Plate prompt (landscape, 1536×1024, gpt-image class model):
"Golden-hour aerial photograph of a curved stone railway viaduct high over a forested Alpine gorge, a red Swiss electric train (locomotive and three red coaches) crossing it, snow-capped peaks lit pink and gold behind under a clear blue sky, the sun rising at the far left horizon with a warm peach glow, a green valley with a village and white church far below on the left with low mist lying in it, a river, conifers in the foreground lower left, a sunlit limestone cliff at the right. 35 mm, drone height, crisp, natural colour, Kodak Portra warmth, real photography. Open sky in the upper left third for a headline. No text, no logos."

Then the same photo edited to remove the train (an image-edit pass: "remove the red train from the viaduct so the empty track, parapet and overhead line continue along the curve, keep everything else identical"). The two are pixel-aligned (phase correlation < 0.2 px).

- `plate.webp`: the empty plate, Lanczos ×1.875 to 2880×1920, unsharp 1.4px 45%, cwebp q86.
- `train.webp` (1387×415, alpha): the train lifted out of the ORIGINAL photo (Apple Vision subject lift on a 4× crop), so its light and grain are the plate's own. It sits at plate (960.5, 492.75) at 4 sprite px per plate px.
- `fg.webp` (1312×1151, alpha): plate pixels cut with painted masks (zones × a foliage key: HSV saturation > .38, hue 35–110°). Rows 0–617 = things at viaduct depth that stand in front of the track (the trees where the track enters the forest at the hairpin, the catenary pole at x 773); rows 618–1150 = the near trees and rock lower left (plate 560–1260 × 740–1024).
- `masks.webp` (768×512): R = depth (0 sky and peaks, 1 everything nearer, feathered 14px), G = where mist may live (five soft ellipses along the valley floor and the slope), B = where cloud may live (two ellipses on the peaks, one on the far ridge).

## Layers (back to front)

1. **Still**: the plate as a CSS background (shown until WebGL draws, kept if WebGL is missing).
2. **WebGL canvas** (one WebGL1 context, DPR ≤ 2, paused off screen). The plate is mapped like `cover` ×1.06 overscan (phone ×1.08, focus x .68 on the viaduct). Fragment shader, ≤ 8 texture reads in any pixel:
   - Parallax: plate uv shifted by mix(far, mid, depth).
   - Mist: fbm of a 256² 4-channel value-noise texture (16 smooth quintic cells per repeat, one octave per channel at ×1, 2.03, 4.11, 8.27), q = (u·.32 − .0034t, v·1.05), domain-warped by one coarse tap, `smoothstep(.38,.68)` × G × .82, colour from cool (.82,.85,.89) to sunlit (1,.92,.82) towards the sun.
   - Cloud: the same at q = (u·.38 + .0026t, v·1.5), `smoothstep(.40,.66)` × B × .78, colour (1,.95,.93).
   - Sun shafts: two taps of the noise along the angle from the sun (−.02, .33), `smoothstep(.48,.82)` × exp(−2.1·dist) below the sun line, + rgb(1,.76,.46) × .2; the sun glow breathes (.09 ± .035, sin .45t).
3. **Train canvas** (2D, DPR ≤ 2). The track is a Catmull-Rom line of 31 hand-placed plate points under the train's near side, each with g (plate px per metre: 6.8 at the near end → 2.3 on the hairpin) and h (train height px: 52 → 25). Distances are integrated in metres, so the train slows and shrinks as it goes away. Each car (locomotive + 7 coaches cycling the three coach cut-outs, 166 m) is cut into 8 vertical strips; each strip is drawn with its own affine matrix that maps its source baseline (where it sat in the photo) and vertical onto the track chord and the local height. Speed 11 m/s × `speed`; trains follow 60 m apart; at 3 s after mount the locomotive sits exactly where the photo had it. Two warm headlamp glows (`lighter`). Then the viaduct-depth occluders from fg.webp.
4. **Near trees** (fg.webp lower part, DOM layer).
5. **Shade** (top 36% → 0 at 42%, bottom 42% dark) and a 9% multiply grain.
6. **Copy, pill, nav.**

Parallax (px at 1440, eased with 1 − e^(−2.6dt)): far 3/2, mid 12/6, near trees 30/12, headline −10/−5 (moves the other way). Ambient dolly with no pointer: x = .42 sin .39t + .18 sin(.17t + 1.3), y = .3 sin(.27t + .6); with a pointer, 85% pointer + 35% dolly.

## Motion timings
