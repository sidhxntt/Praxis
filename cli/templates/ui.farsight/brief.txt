```tsx
"use client";

/**
 * FARSIGHT · website hero
 *
 * The first screen of a magazine of places photographed from very far away.
 * On a black page, under a thin serif headline, seven tall photographs stand
 * on a ring in real 3D: the one in the middle faces you, its neighbours turn
 * away in perspective and the outer two are slivers. The pictures are the
 * light of the page: each one spills its own colour onto the floor under it,
 * is mirrored there, and the one in the middle lights the room behind it.
 *
 * The ring never stops. It glides so every picture takes the middle in turn
 * (slow as it arrives, quick between two), the one arriving opens a little
 * wider, its caption rises, its photograph slides inside its frame like a
 * view through a window, and a band of light crosses each card as it turns.
 * The counter in the corner rolls with it.
 *
 * The pointer leans the whole ring and tilts the middle card. Drag the row to
 * spin it (throw it and it carries on), click a side card to bring it in, or
 * use the arrow keys. After a touch the ring waits six seconds, then goes on.
 *
 * One file, React only. The typefaces come from Google Fonts (loaded in the CSS).
 *
 *   <Farsight />
 *   <Farsight pace={1.4} glow={1.2} onOpen={(place) => router.push(`/places/${place.slug}`)} />
 *
 * Props
 *   brand    the wordmark                                        "Farsight"
 *   cta      the button in the nav                               "Subscribe"
 *   places   the pictures: { slug, src, title: [line, line], hue }  seven aerial photographs
 *   pace     how fast the ring travels (1 = a picture every 2.2 s)  1
 *   glow     how much light the pictures throw (0.4 to 1.6)       1
 *   turn     how far the side cards turn away (0.7 to 1.2)        1
 *   sound    a tiny tick as a picture passes under your hand       true
 *   onCta    called when the nav button is pressed
 *   onOpen   called with the place when the middle card is pressed
 *
 * Fills its container's width; height is var(--design-h, 100svh), never under
 * 640px. Everything is sized from one unit that follows the smaller of the
 * width and the height, so it holds from a phone to a cinema screen. On a
 * phone (container under 640px) the headline stacks above one card and two
 * slivers. The ring pauses off screen and in a hidden tab. Reduced motion: no
 * travel and no drift; drag, click and keys still move it, at once.
 * Keyboard: Tab to the row, Left/Right step, Home/End jump, Enter opens.
 * Sound: silent until the visitor interacts, when `sound` is false, and when
 * <html data-cd-mute="1">. The AudioContext is created on the first tick.
 */

import { useCallback, useEffect, useRef, type CSSProperties, type KeyboardEvent as ReactKeyboardEvent, type PointerEvent as ReactPointerEvent } from "react";

export type Place = { slug: string; src: string; title: [string, string]; hue: string };

type Props = {
  brand?: string;
  cta?: string;
  places?: Place[];
  pace?: number;
  glow?: number;
  turn?: number;
  sound?: boolean;
  onCta?: () => void;
  onOpen?: (place: Place) => void;
};

const DIR = "/design/websites/farsight/";
const PLACES: Place[] = [
  { slug: "delta", src: DIR + "delta.webp", title: ["River deltas that", "look like lungs"], hue: "#35bfb0" },
  { slug: "salt", src: DIR + "salt.webp", title: ["Salt pans, sorted", "by colour"], hue: "#ff4d7d" },
  { slug: "glacier", src: DIR + "glacier.webp", title: ["A glacier, one day", "from the sea"], hue: "#6fc9f0" },
  { slug: "dunes", src: DIR + "dunes.webp", title: ["Sand seas and", "their slow waves"], hue: "#ff7a1f" },
  { slug: "reef", src: DIR + "reef.webp", title: ["Reefs drawn", "in turquoise"], hue: "#12c8c4" },
  { slug: "fields", src: DIR + "fields.webp", title: ["Fields,", "as a quilt"], hue: "#d6d21f" },
  { slug: "city", src: DIR + "city.webp", title: ["Cities", "at 3 a.m."], hue: "#ffb14a" },
];

/* Where a card stands, by its distance from the middle (0, 1, 2, 3 cards away).
   Units: x and z in the layout unit (see --fsg-u), turn in degrees. */
const XT = [0, 27, 49, 59];
const ZT = [0, -5, -17, -30];
const RT = [0, 28, 48, 64];
const OT = [1, 1, 0.92, 0]; // how visible
const DT = [0, 0.14, 0.34, 0.66]; // how much shade lies on it
const PERSP = 112; // the camera's distance, in units
const STEP = 2.2; // seconds a picture, at pace 1
const SLOW = 0.8; // how much the ring slows as a picture arrives (0 = steady, 1 = it would stop)
const WAIT = 6; // seconds the ring waits after a touch

const lerpT = (t: number[], a: number) => {
  const i = Math.min(t.length - 2, Math.floor(a));
  const f = Math.min(1, a - i);
  return t[i] + (t[i + 1] - t[i]) * f;
};
const r2 = (n: number) => Math.round(n * 100) / 100;
const r3 = (n: number) => Math.round(n * 1000) / 1000;

/* One card's whole look for a ring position. Strings only, so the server and the browser agree. */
function pose(i: number, n: number, pos: number, t: number, px: number, py: number, hov: number, turn: number, glow: number) {
  let d = (((i - pos) % n) + n) % n;
  if (d > n / 2) d -= n;
  const a = Math.min(3, Math.abs(d));
  const s = d < 0 ? -1 : 1;
  const w0 = Math.max(0, 1 - a);
  const w = w0 * w0 * (3 - 2 * w0); // 1 in the middle, 0 a card away
  const x = s * lerpT(XT, a);
  const z = lerpT(ZT, a);
  const ry = s * lerpT(RT, a) * turn + px * 7 * w * hov;
  const rx = -py * 5 * w * hov;
  const op = lerpT(OT, a);
  const sx = 0.9 + 0.1 * w; // the middle card opens wider
  const k = PERSP / (PERSP - z); // how big it is drawn at that depth
  const ph = i * 1.7;
  const ix = -d * 5.2 + Math.sin(t * 0.21 + ph) * 1.6; // the photograph slides inside its frame
  const iy = Math.cos(t * 0.17 + ph) * 2.2 - w * 1.2;
  const is = 1.16 + 0.07 * w + Math.sin(t * 0.13 + ph) * 0.015;
  return {
    card: `translateX(calc(var(--fsg-u) * var(--fsg-kx) * ${r2(x)})) translateZ(calc(var(--fsg-u) * ${r2(z)})) rotateY(calc(var(--fsg-kr) * ${r2(ry)}deg)) rotateX(${r2(rx)}deg) scale(${r3(1 + 0.035 * w)})`,
    op: r3(op),
    face: `scaleX(${r3(sx)})`,
    img: `scaleX(${r3(1 / sx)}) translate3d(${r2(ix)}%, ${r2(iy)}%, 0) scale(${r3(is)})`,
    shade: r3(lerpT(DT, a)),
    cap: `scaleX(${r3(1 / sx)}) translateY(calc(var(--fsg-u) * ${r2((1 - w) * 0.9)}))`,
    capOp: r3(0.62 + 0.38 * w),
    sheen: `translateX(${r2(-d * 150 + px * 26 * hov)}%)`,
    sheenOp: r3(Math.max(0, 1 - Math.abs(d * 1.5)) * 0.6 + 0.14 * w * hov),
    pool: `translateX(calc(var(--fsg-u) * var(--fsg-kx) * ${r2(x * k)})) scale(${r3(k * (0.5 + 0.5 * Math.cos((Math.min(80, Math.abs(ry)) * Math.PI) / 180)) * (1 + 0.25 * w))}, ${r3(k)})`,
    poolOp: r3(Math.min(1, op * (0.5 + 0.5 * w) * glow)),
    room: r3(Math.min(1, w * 0.95 * glow)),
    front: w > 0.5,
  };
}

/* Sound: one tiny tick, made in the browser. Silent when `sound` is false, when
   <html data-cd-mute="1">, and until the visitor has interacted. */
function useTick(on: boolean) {
  const ctx = useRef<AudioContext | null>(null);
  useEffect(() => () => void ctx.current?.close().catch(() => {}), []);
  return useCallback(
    (high = false) => {
      if (!on || typeof window === "undefined") return;
      if (document.documentElement.dataset.cdMute === "1") return;
      const ua = (navigator as unknown as { userActivation?: { hasBeenActive: boolean } }).userActivation;
      if (ua && !ua.hasBeenActive) return;
      try {
        const AC = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
        if (!AC) return;
        const a = (ctx.current ??= new AC());
        if (a.state === "suspended") void a.resume();
        const t = a.currentTime;
        const o = a.createOscillator();
        const g = a.createGain();
        o.type = "triangle";
        o.frequency.setValueAtTime(high ? 1320 : 880, t);
        o.frequency.exponentialRampToValueAtTime(high ? 1760 : 620, t + 0.05);
        g.gain.setValueAtTime(0.0001, t);
        g.gain.exponentialRampToValueAtTime(0.035, t + 0.004);
        g.gain.exponentialRampToValueAtTime(0.0001, t + 0.07);
        o.connect(g).connect(a.destination);
        o.start(t);
        o.stop(t + 0.08);
      } catch {
        /* no sound is fine */
      }
    },
    [on],
  );
}

export default function Farsight({ brand = "Farsight", cta = "Subscribe", places = PLACES, pace = 1, glow = 1, turn = 1, sound = true, onCta, onOpen }: Props) {
  const n = places.length;
  const root = useRef<HTMLDivElement>(null);
  const ruler = useRef<HTMLSpanElement>(null);
  const roll = useRef<HTMLSpanElement>(null);
  const live = useRef<HTMLSpanElement>(null);
  const ring = useRef<HTMLDivElement>(null);
  const head = useRef<HTMLDivElement>(null);
  const cards = useRef<(HTMLButtonElement | null)[]>([]);
  const pools = useRef<(HTMLSpanElement | null)[]>([]);
  const rooms = useRef<(HTMLSpanElement | null)[]>([]);
  const tick = useTick(sound);

  /* Everything that moves lives here, outside React: the loop writes styles straight to the elements. */
  const st = useRef({ pos: 0, goal: null as number | null, t: 0, px: 0, py: 0, tx: 0, ty: 0, hov: 0, hovT: 0, touched: -1e9, drag: false, lastX: 0, lastT: 0, v: 0, downX: 0, downI: -1, moved: false, front: 0, still: false });
  const conf = useRef({ pace, glow, turn, n });
  conf.current = { pace, glow, turn, n };
  const cb = useRef({ onOpen, places });
  cb.current = { onOpen, places };

  const paint = useCallback(() => {
    const s = st.current;
    const c = conf.current;
    for (let i = 0; i < c.n; i++) {
      const el = cards.current[i];
      if (!el) continue;
      const p = pose(i, c.n, s.pos, s.t, s.px, s.py, s.hov, c.turn, c.glow);
      el.style.transform = p.card;
      el.style.opacity = String(p.op);
      el.style.pointerEvents = p.op < 0.3 ? "none" : "";
      const k = el.children as HTMLCollectionOf<HTMLElement>;
      // children: halo, face, mirror
      k[0].style.transform = p.face;
      k[1].style.transform = p.face;
      k[2].style.transform = p.face;
      const f = k[1].children as HTMLCollectionOf<HTMLElement>; // img, shade, sheen, caption
      f[0].style.transform = p.img;
      f[1].style.opacity = String(p.shade);
      f[2].style.transform = p.sheen;
      f[2].style.opacity = String(p.sheenOp);
      f[3].style.transform = p.cap;
      f[3].style.opacity = String(p.capOp);
      (k[2].firstElementChild as HTMLElement).style.transform = p.img;
      const pool = pools.current[i];
      if (pool) {
        pool.style.transform = p.pool;
        pool.style.opacity = String(p.poolOp);
      }
      const room = rooms.current[i];
      if (room) room.style.opacity = String(p.room);
    }
    if (roll.current) roll.current.style.transform = `translateY(${r3(-((((s.pos % c.n) + c.n) % c.n) / (c.n + 1)) * 100)}%)`;
    if (ring.current) ring.current.style.transform = `rotateY(${r2(s.px * 3.2 + Math.sin(s.t * 0.31) * 0.9)}deg) rotateX(${r2(-s.py * 1.6)}deg)`;
    if (head.current) head.current.style.transform = `translate3d(calc(var(--fsg-u) * ${r2(-s.px * 0.5)}), calc(var(--fsg-u) * ${r2(-s.py * 0.3)}), 0)`;
    const front = ((Math.round(s.pos) % c.n) + c.n) % c.n;
    if (front !== s.front) {
      s.front = front;
      // the keyboard's place follows the picture in the middle
      const held = cards.current.some((el) => el !== null && el === document.activeElement);
      cards.current.forEach((el, i) => {
        if (!el) return;
        el.tabIndex = i === front ? 0 : -1;
        el.setAttribute("aria-current", i === front ? "true" : "false");
      });
      if (held) cards.current[front]?.focus({ preventScroll: true });
      const pl = cb.current.places[front];
      if (live.current && pl) live.current.textContent = `${front + 1} of ${c.n}: ${pl.title.join(" ")}`;
      return true;
    }
    return false;
  }, []);

  useEffect(() => {
    const el = root.current;
    if (!el) return;
    const s = st.current;
    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    s.still = mq.matches;
    const onMq = () => {
      s.still = mq.matches;
    };
    mq.addEventListener("change", onMq);
    let raf = 0;
    let seen = true;
    let last = performance.now();
    const loop = (now: number) => {
      raf = 0;
      const dt = Math.min(0.05, (now - last) / 1000);
      last = now;
      const c = conf.current;
      if (!s.still) s.t += dt;
      // the pointer, eased
      const e = 1 - Math.exp(-dt * 6);
      s.px += (s.tx - s.px) * e;
      s.py += (s.ty - s.py) * e;
      s.hov += (s.hovT - s.hov) * e;
      const quiet = now / 1000 - s.touched;
      if (s.drag) {
        /* in the hand: the pointer handler moves pos */
      } else if (s.goal !== null) {
        if (s.still) s.pos = s.goal;
        else s.pos += (s.goal - s.pos) * (1 - Math.exp(-dt * 6.5));
        if (Math.abs(s.goal - s.pos) < 0.002) {
          s.pos = s.goal;
          s.goal = null;
        }
      } else if (!s.still && quiet > WAIT) {
        // travelling by itself: slow as a picture arrives, quick between two, never still
        const ramp = Math.min(1, (quiet - WAIT) / 1.2);
        const period = 1 / Math.sqrt(1 - SLOW * SLOW);
        const v = ((1 - SLOW * Math.cos(2 * Math.PI * s.pos)) * period) / (STEP / Math.max(0.2, c.pace));
        s.pos += v * dt * ramp;
      }
      const changed = paint();
      if (changed && quiet < WAIT) tick();
      if (seen && document.visibilityState === "visible") raf = requestAnimationFrame(loop);
    };
    const start = () => {
      if (!raf && seen && document.visibilityState === "visible") {
        last = performance.now();
        raf = requestAnimationFrame(loop);
      }
    };
    const io = new IntersectionObserver(([x]) => {
      seen = x.isIntersecting;
      start();
    });
    io.observe(el);
    document.addEventListener("visibilitychange", start);
    start();
    return () => {
      cancelAnimationFrame(raf);
      io.disconnect();
      document.removeEventListener("visibilitychange", start);
      mq.removeEventListener("change", onMq);
    };
  }, [paint, tick]);

  const touch = () => {
    st.current.touched = performance.now() / 1000;
  };
  /* one card's width on screen, in pointer pixels (the stage may be shown scaled) */
  const cardPx = () => Math.max(40, ruler.current?.getBoundingClientRect().width ?? 320);

  const onMove = (e: ReactPointerEvent<HTMLDivElement>) => {
    const s = st.current;
    const r = e.currentTarget.getBoundingClientRect();
    s.tx = Math.max(-1, Math.min(1, ((e.clientX - r.left) / r.width) * 2 - 1));
    s.ty = Math.max(-1, Math.min(1, ((e.clientY - r.top) / r.height) * 2 - 1));
    s.hovT = e.pointerType === "mouse" ? 1 : 0;
  };
  const onLeave = () => {
    const s = st.current;
    s.tx = 0;
    s.ty = 0;
    s.hovT = 0;
  };

  const down = (e: ReactPointerEvent<HTMLDivElement>) => {
    if (e.button !== 0) return;
    const s = st.current;
    const hit = (e.target as HTMLElement).closest<HTMLElement>("[data-fsg-i]");
    s.drag = true;
    s.moved = false;
    s.goal = null;
    s.v = 0;
    s.downX = s.lastX = e.clientX;
    s.lastT = performance.now();
    s.downI = hit ? Number(hit.dataset.fsgI) : -1;
    touch();
    try {
      e.currentTarget.setPointerCapture(e.pointerId);
    } catch {
      /* fine */
    }
  };
  const move = (e: ReactPointerEvent<HTMLDivElement>) => {
    const s = st.current;
    if (!s.drag) return;
    const dx = e.clientX - s.lastX;
    if (Math.abs(e.clientX - s.downX) > 6) s.moved = true;
    if (!s.moved) return;
    const now = performance.now();
    const dp = -dx / cardPx();
    s.pos += dp;
    const dtt = Math.max(1, now - s.lastT) / 1000;
    s.v = s.v * 0.6 + (dp / dtt) * 0.4;
    s.lastX = e.clientX;
    s.lastT = now;
    touch();
  };
  const up = (e: ReactPointerEvent<HTMLDivElement>) => {
    const s = st.current;
    if (!s.drag) return;
    s.drag = false;
    try {
      e.currentTarget.releasePointerCapture(e.pointerId);
    } catch {
      /* fine */
    }
    touch();
    if (s.moved) {
      // thrown: it carries on for a moment, then comes to rest on a picture
      const stale = performance.now() - s.lastT > 90 ? 0 : s.v;
      s.goal = Math.round(s.pos + Math.max(-3, Math.min(3, stale * 0.28)));
    } else if (s.downI >= 0) {
      bring(s.downI);
    }
  };

  const bring = (i: number) => {
    const s = st.current;
    const c = conf.current;
    const base = Math.round(s.pos);
    let d = (((i - base) % c.n) + c.n) % c.n;
    if (d > c.n / 2) d -= c.n;
    touch();
    if (d === 0) {
      s.goal = base;
      tick(true);
      const pl = cb.current.places[i];
      if (pl) cb.current.onOpen?.(pl);
      // a nod, so the press is seen: the middle card dips and comes back
      const el = cards.current[i]?.children[1] as HTMLElement | undefined;
      if (el && !s.still) {
        el.classList.remove("fsg-nod");
        void el.offsetWidth;
        el.classList.add("fsg-nod");
      }
    } else {
      s.goal = base + d;
    }
  };
  const onKey = (e: ReactKeyboardEvent<HTMLDivElement>) => {
    const s = st.current;
    const c = conf.current;
    const base = s.goal ?? Math.round(s.pos);
    if (e.key === "ArrowRight" || e.key === "ArrowLeft") {
      e.preventDefault();
      s.goal = base + (e.key === "ArrowRight" ? 1 : -1);
      touch();
    } else if (e.key === "Home" || e.key === "End") {
      e.preventDefault();
      const cur = ((base % c.n) + c.n) % c.n;
      s.goal = base - cur + (e.key === "Home" ? 0 : c.n - 1);
      touch();
    }
  };
  /* a keyboard press on a card (Enter/Space on a button arrives as a click with no pointer) */
  const onCardClick = (i: number) => (e: { detail: number }) => {
    if (e.detail === 0) bring(i);
  };

  const vars = { "--fsg-glow": String(glow) } as CSSProperties;
  const first = places.map((_, i) => pose(i, n, 0, 0, 0, 0, 0, turn, glow));
  const pad = (k: number) => String(k).padStart(2, "0");

  return (
    <div ref={root} className="fsg-root" style={vars} onPointerMove={onMove} onPointerLeave={onLeave}>
      <style>{css}</style>
      <div className="fsg-stage">
        <nav className="fsg-nav" aria-label="Farsight">
          <a className="fsg-brand" href="#" onClick={(e) => e.preventDefault()}>
            {brand}
            <i aria-hidden />
          </a>
          <div className="fsg-links">
            {["Issues", "Places", "Photographers", "About"].map((l) => (
              <a key={l} href="#" onClick={(e) => e.preventDefault()}>
                {l}
              </a>
            ))}
            <button type="button" className="fsg-cta" onClick={() => onCta?.()}>
              {cta}
            </button>
          </div>
        </nav>

        <div className="fsg-comp">
          <span ref={ruler} className="fsg-ruler" aria-hidden />
          <div className="fsg-rooms" aria-hidden>
            {places.map((p, i) => (
              <span key={p.slug} ref={(el) => void (rooms.current[i] = el)} style={{ "--fsg-hue": p.hue, opacity: first[i].room } as CSSProperties} />
            ))}
          </div>

          <div ref={head} className="fsg-head">
            <h1 className="fsg-h1">
              <span>Seven places</span>
              <span>you can only see</span>
              <span>from far away</span>
            </h1>
          </div>

          <div className="fsg-floor" aria-hidden />
          <div className="fsg-pools" aria-hidden>
            {places.map((p, i) => (
              <span key={p.slug} ref={(el) => void (pools.current[i] = el)} style={{ "--fsg-hue": p.hue, transform: first[i].pool, opacity: first[i].poolOp } as CSSProperties} />
            ))}
          </div>

          <div
            className="fsg-row"
            role="group"
            aria-roledescription="carousel"
            aria-label="Seven places. Drag, or use the arrow keys, to turn the row"
            onPointerDown={down}
            onPointerMove={move}
            onPointerUp={up}
            onPointerCancel={up}
            onKeyDown={onKey}
          >
            <div ref={ring} className="fsg-ring">
              {places.map((p, i) => (
                <button
                  key={p.slug}
                  type="button"
                  ref={(el) => void (cards.current[i] = el)}
                  className="fsg-card"
                  data-fsg-i={i}
                  tabIndex={i === 0 ? 0 : -1}
                  aria-current={i === 0 ? "true" : "false"}
```

The full brief of this design, for reference, or to rebuild it in another stack:

# Farsight · magazine hero with a ring of photographs

Build the first screen of "Farsight", a quarterly magazine of places photographed from very far away (orbit, aircraft). The page is black and the pictures are its only light: seven tall aerial photographs stand on a ring in real 3D under a thin serif headline. The one in the middle faces the reader, its neighbours turn away, the outer two are slivers. The ring never stops turning, so every picture takes the middle in turn. One screen, no scroll.

## Pictures (seven, portrait 2:3, 640×960 webp, about 100 to 290 KB each)

One prompt tail for all of them: "Real aerial photograph, true nadir view straight down from very high altitude, vertical portrait frame filled edge to edge, vivid saturated natural colour, crisp fine detail, slight film grain, no horizon, no sky, no text, no labels, no border, no people, no overlay graphics."

1. `delta` (glow `#35bfb0`): "A branching river delta seen from orbit: dozens of teal and turquoise water channels dividing like the bronchi of a lung through rust-red and ochre silt flats, with fine dark green veins of mangrove along the channels and milky sediment plumes where they meet a deep blue sea at the bottom of the frame."
2. `salt` (`#ff4d7d`): "Salt evaporation ponds seen from above: a mosaic of large geometric ponds in vivid magenta pink, coral, rust orange, lime green, emerald and milky white, separated by thin pale straight dikes, with fine crystalline salt crusts and swirling mineral patterns inside the ponds."
3. `glacier` (`#6fc9f0`): "A glacier tongue seen from high above: a wide river of luminous blue-white ice with long curved dark moraine stripes and fine crevasse patterns, flowing from the top of the frame between dark charcoal rock walls dusted with snow, ending in a milky turquoise meltwater lake with small white icebergs at the bottom."
4. `dunes` (`#ff7a1f`): "A sand sea seen from orbit: endless long curved orange and deep amber dunes with razor-sharp crests, one side of every dune glowing warm orange in low sun and the other in deep violet-brown shadow, fine wind ripples, a few pale flat salt pans between the dune chains."
5. `reef` (`#12c8c4`): "A coral reef and atoll seen from high above: a ring of pale sand and electric turquoise shallow lagoon with darker teal coral patterns and veins, surrounded by deep saturated cobalt blue ocean, small white wave lines breaking on the outer reef edge."
6. `fields` (`#d6d21f`): "A patchwork of farm fields seen from very high: hundreds of small irregular plots in bright yellow rapeseed, emerald green, lime, terracotta ploughed earth and purple lavender, stitched together by thin pale tracks and dark hedgerows, a few perfect green crop circles, one silver river curling through."
7. `city` (`#ffb14a`): "A dense city at night seen from orbit straight down: a glowing web of amber-gold and warm white street lights, bright arterial roads radiating from a brilliant centre, cool blue-white districts, a dark river with lit bridges winding through, black unlit parks and water. Sharp points of light, deep black between them."

Pick each by one test: at thumbnail size it reads as an abstract painting, at full size as a real photograph. Each picture has one glow colour (above), taken from its most saturated third.

## The unit

Everything is sized from one unit `u = min(1% of the width, 1.6% of the height)`, so 1440×900 gives `u = 14.4px` and a shorter screen shrinks the whole composition instead of cropping it. The composition is a box of 100u × 62.5u centred on the screen; only the nav is pinned to the real edges. All numbers below are in `u` unless they say px.

## Layers (back to front)

1. **Ground**: `#000`.
2. **Room light**: seven stacked radial gradients (86 × 56, centred at y = 41), one per picture: its glow colour at 42% in the middle, 17% at 42% of the radius, transparent at the edge. The opacity of each is the weight `w` of its picture (1 when it is in the middle, 0 one card away), so the room changes colour as the ring turns.
3. **Headline** (centred, top at 8.9): three lines, "Seven places / you can only see / from far away". Fraunces 300 at optical size 144, 3.72 (53.5px), line-height 1.085, tracking −0.022em, white.
4. **Floor**: a strip 6.3 high from y = 56.2, white 5.5% fading down, faded out at both ends. On it, one **pool** of colour under every card: a 34 × 9.2 radial gradient (the glow colour at 70% lifted 8% toward white in the middle, 38% at a third, transparent), blended `screen`, placed at the card's x after perspective and squeezed as the card turns.
5. **The ring** (a 3D context: perspective 112, origin 50% 42%, the row starts at y = 24.6). Seven cards, each 23.2 × 31 with a 1.55 radius. A card that is `a` cards from the middle (the ring position is a real number, so `a` is too) stands at:

   | a | x | z | turned | visible | shade |
   |---|---|---|---|---|---|
   | 0 | 0 | 0 | 0° | 1 | 0 |
   | 1 | 27 | −5 | 28° | 1 | 14% black |
   | 2 | 49 | −17 | 48° | 0.92 | 34% |
   | 3 | 59 | −30 | 64° | 0 | 66% |

   Interpolate straight between rows. Cards on the right turn their right edge away, cards on the left their left edge (a ring seen from outside). With `w = smoothstep(1 − a)`:
   - the card is scaled by `1 + 0.035w` and its face is `0.9 + 0.1w` as wide (the picture and the caption are counter-scaled, so only the frame opens);
   - the photograph is bigger than its frame (scale `1.16 + 0.07w`) and slides inside it by `−5.2% × d` (d is the signed distance), plus a slow drift of ±1.6% and ±2.2% on two sine clocks of about 30 s and 37 s with a different phase per card: like a view through a window;
   - a band of light (a 100° white gradient, peak 62%, blended `screen`) crosses the card as it passes the middle: its x is `−150% × d`, its opacity `0.6 × max(0, 1 − 1.5|d|)`;
   - the caption (two centred lines at the foot, Fraunces 400, 1.5 (21.6px), line-height 1.08, on a black gradient to 62%) rises 0.9 and goes from 62% to full white as the card arrives;
   - behind the face a halo of the glow colour (36%), and under it the **mirror**: the same picture upside down, 0.42 below the card, 50% opacity, masked from 62% at the card's edge to nothing at 21% of its height.
   - hairline: inset 1px white 16%, with a brighter top edge (28%).
   The whole ring also sways (±0.9° on a 20 s sine) and leans with the pointer (up to 3.2° around y, 1.6° around x).
6. **UI** (Inter Tight)
   - Nav, 2.9 from the top, 4.1 from the sides: "Farsight" 1.16 (16.7px) 600 followed by a white dash 1.9 long; links Issues, Places, Photographers, About at 0.98 (14px) 500, white 86%; a white pill "Subscribe" 0.98 600, ink `#0a0a0a`, padding 0.72 × 1.35.
   - Bottom left (4.1 in, 2.5 up, 21 wide): "**Issue 14** · Winter. A quarterly of places photographed from orbit, from aircraft and from very tall ladders." 0.92 (13.2px), line-height 1.38, white 56%, "Issue 14" in white 600.
   - Bottom right: a rolling counter "02 — 07" in tabular figures (the first number is a column of 01…07,01 moved by the ring position, so it rolls exactly as the ring turns), then a hairline divider and "Drag to turn". 0.98, white 56%, the live number white.

## Motion

- **Travel**: the ring position `p` obeys `dp/dt = (1 − 0.8·cos 2πp) · 1.667 / 2.2 s`. It never stops: it crawls as a picture reaches the middle (about 0.15 cards a second) and runs between two (about 1.36), one picture every 2.2 s. `pace` divides the 2.2.
- **In the hand**: dragging the row moves `p` by the pointer's travel divided by one card step (27u, in screen pixels). On release it goes on to `round(p + 0.28 × speed)` (at most three cards), easing with `1 − e^(−6.5·dt)`. A click on a side card brings it to the middle; a click on the middle card opens it (a short dip of the card, 0.5 s, and the `onOpen` callback).
- **Pointer**: its position in the hero (−1…1) is eased with `1 − e^(−6·dt)`; it leans the ring, tilts the middle card (7° and 5°), nudges the band of light, and moves the headline against it by 0.5 × 0.3.
- **After a touch** (drag, click, key) the travel waits 6 s, then comes back over 1.2 s.
- **Entrance**: the three headline lines rise 1.2 and fade in over 0.9 s, 70 ms apart. The ring, the light and the floor are there from the first frame.
- **Sound**: one triangle tick (880 → 620 Hz, 70 ms, gain 0.035) each time a new picture reaches the middle while the visitor is driving; a higher one (1320 → 1760 Hz) on open. Never during the travel by itself, never before a user gesture.
- Paused off screen and in a hidden tab. Reduced motion: no travel, no drift, no entrance; drag, click and keys move the ring at once.

## Keyboard

One tab stop in the row: the card in the middle (it moves with the ring, and focus follows it). Left/Right step, Home/End go to the first and last, Enter or Space opens. Focus ring: 2px black then 2px white around the card. A polite live region reads "3 of 7: A glacier, one day from the sea".

## Phone (container under 640px)

`u = min(2.564% of the width, 1.185% of the height)` (10px at 390×844), composition 39 × 84.4. Nav: wordmark 1.7 and the pill only. Headline at y = 15.2, size 4.25. The row starts at 32.2 with perspective 78; cards 25.2 × 34.4; the side cards stand 0.6 as far apart and turn 2.15 times as much, so the neighbours are slivers at the screen's edges and the next ones face away and are hidden (`backface-visibility: hidden`). Note and counter at the bottom, 1.22 and 1.3; "Drag to turn" is dropped.
