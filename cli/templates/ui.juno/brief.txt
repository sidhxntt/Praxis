/**
 * JUNO · website hero
 *
 * The first screen of a product designer's portfolio, made as a page of her
 * notebook: pale lined paper, a tiny pixel nav, "my name is" in handwriting,
 * and the name set in a chunky pixel face inside a design tool's selection
 * frame with its eight handles. Washi-tape labels point at it with drawn
 * arrows, three taped polaroids sit at the edges, three torn notes carry her
 * numbers, and a pixel smiley sticker waits in the corner.
 *
 * The page is a desk you can touch. Everything taped or framed can be picked
 * up: it tilts as it moves, casts a deeper shadow, and springs home when you
 * let go. The polaroids hang from their tape, so they swing like a pendulum
 * and settle. Drag a handle of the frame and the name stretches with it like
 * in a design tool (the size readout counts), then eases back. Point at a
 * tape label and it peels up from one end.
 *
 * Left alone, a hand you cannot see does it for you: a polaroid is pulled and
 * let go, the frame is stretched, the labels lift one after another, the
 * smiley winks. It stops for six seconds after any real touch.
 *
 * One file, React only. The typefaces come from Google Fonts (loaded in the
 * CSS); the name itself is drawn from a small pixel alphabet in this file, so
 * it can stretch without blurring.
 *
 *   <Juno />
 *   <Juno name="MIRA" frame="#2f6bff" paper="grid" />
 *
 * Props
 *   name      the name in the frame (A–Z, 0–9)            "JUNO"
 *   line      the statement, two lines                    ["I make software you stop noticing", "because it just works."]
 *   cta       the button                                  "See the work"
 *   frame     the selection frame's colour                "#f0452e"
 *   paper     "lined" | "grid" | "plain"                  "lined"
 *   portrait  polaroid 1 (square)                         "/design/websites/juno/juno.webp"
 *   street    polaroid 2 (square)                         "/design/websites/juno/tram.webp"
 *   desk      polaroid 3 (square)                         "/design/websites/juno/desk.webp"
 *   sound     tiny synthesized paper sounds               true
 *   onCta     called when the button is pressed
 *
 * Fills its container's width; height is var(--design-h, 100svh), never under
 * 640px. Responsive by container query: on a phone the nav shortens, the
 * statement breaks into three lines and two polaroids overlap at the bottom.
 * Everything pauses off screen and in a hidden tab.

/* ------------------------------------------------------------------ sound
   A tiny synthesized tick or tap, made in the browser. Silent when `sound`
   is false, when <html data-cd-mute="1">, and until the visitor has
   interacted. Delete this hook and its calls to make it silent. */
type Blip = { f: number; to?: number; d?: number; g?: number; noise?: boolean; q?: number };
function useBlip(on: boolean) {
  const ctx = useRef<AudioContext | null>(null);
  useEffect(() => () => void ctx.current?.close().catch(() => {}), []);
  return useCallback(
    (b: Blip) => {
      if (!on || typeof window === "undefined") return;
      if (document.documentElement.dataset.cdMute === "1") return;
      const ua = (navigator as unknown as { userActivation?: { hasBeenActive: boolean } }).userActivation;
      if (ua && !ua.hasBeenActive) return;
      try {
        const AC = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
        if (!AC) return;
        const a = (ctx.current ??= new AC());
        if (a.state === "suspended") void a.resume();
        const t = a.currentTime;
        const d = Math.min(b.d ?? 0.04, 0.06);
        const amp = a.createGain();
        amp.gain.setValueAtTime(0.0001, t);
        amp.gain.exponentialRampToValueAtTime(Math.min(b.g ?? 0.03, 0.06), t + 0.005);
        amp.gain.exponentialRampToValueAtTime(0.0001, t + d);
        amp.connect(a.destination);
        if (b.noise) {
          const buf = a.createBuffer(1, Math.ceil(a.sampleRate * d), a.sampleRate);
          const ch = buf.getChannelData(0);
          for (let i = 0; i < ch.length; i++) ch[i] = Math.random() * 2 - 1;
          const n = a.createBufferSource();
          n.buffer = buf;
          const bp = a.createBiquadFilter();
          bp.type = "bandpass";
          bp.frequency.setValueAtTime(b.f, t);
          if (b.to) bp.frequency.exponentialRampToValueAtTime(b.to, t + d);
          bp.Q.value = b.q ?? 0.9;
          n.connect(bp).connect(amp);
          n.start(t);
          n.stop(t + d + 0.01);
        } else {
          const o = a.createOscillator();
          o.type = "triangle";
          o.frequency.setValueAtTime(b.f, t);
          if (b.to) o.frequency.exponentialRampToValueAtTime(b.to, t + d);
          o.connect(amp);
          o.start(t);
          o.stop(t + d + 0.01);
        }
      } catch {
        /* no sound is fine */
      }
    },
    [on],
  );
}

type Run = { x: number; y: number; w: number };
function pixelWord(word: string): { w: number; runs: Run[] } {
  const runs: Run[] = [];
  let x = 0;
  for (const ch of word.toUpperCase().slice(0, 14)) {
    const g = GLYPHS[ch] ?? GLYPHS["-"];
    const gw = g[0].length;
    for (let r = 0; r < 9; r++) {
      let c = 0;
      while (c < gw) {
        if (g[r][c] === "#") {
          let e = c;
          while (e < gw && g[r][e] === "#") e++;
          runs.push({ x: x + c, y: r, w: e - c });
          c = e;
        } else c++;
      }
    }
    x += gw + 1;
  }
  return { w: Math.max(1, x - 1), runs };
}

/* What the unseen hand does, in seconds from the start of a 14 s loop.
   Distances are fractions of the page width. */
type Step =
  | { at: number; dur: number; hold: number; k: string; to: [number, number] }
  | { at: number; dur: number; hold: number; handle: string; to: [number, number] }
  | { at: number; peel: true }
  | { at: number; wink: true };
const SCRIPT: Step[] = [
  { at: 0.7, dur: 1.0, hold: 0.25, k: "p2", to: [-0.1, 0.06] },
  { at: 3.1, dur: 0.85, hold: 0.3, handle: "se", to: [0.085, 0.032] },
  { at: 5.5, peel: true },
  { at: 7.5, wink: true },
  { at: 8.3, dur: 0.9, hold: 0.2, k: "p1", to: [0.075, -0.06] },
  { at: 10.3, dur: 0.8, hold: 0.2, k: "n2", to: [0.04, -0.05] },
  { at: 11.7, dur: 0.7, hold: 0.25, handle: "w", to: [-0.06, 0] },
];
const LOOP = 14;

type Item = { el: HTMLElement; hang: boolean; x: number; y: number; vx: number; vy: number; th: number; om: number; held: 0 | 1 | 2; tx: number; ty: number; rest: boolean };
type Side = { v: number; vel: number; t: number };
const HANDLES: Record<string, [number, number]> = { nw: [-1, -1], n: [0, -1], ne: [1, -1], e: [1, 0], se: [1, 1], s: [0, 1], sw: [-1, 1], w: [-1, 0] };
const easeIO = (p: number) => (p < 0.5 ? 2 * p * p : 1 - Math.pow(-2 * p + 2, 2) / 2);
const clamp = (v: number, a: number, b: number) => Math.min(b, Math.max(a, v));

export default function Juno({
  name = "JUNO",
  line = ["I make software you stop noticing", "because it just works."],
  cta = "See the work",
  frame = "#f0452e",
  paper = "lined",
  portrait = "/design/websites/juno/juno.webp",
  street = "/design/websites/juno/tram.webp",
  desk = "/design/websites/juno/desk.webp",
  sound = true,
  onCta,
}: Props) {
  const root = useRef<HTMLDivElement>(null);
  const frameEl = useRef<HTMLDivElement>(null);
  const readout = useRef<HTMLSpanElement>(null);
  const blip = useBlip(sound);
  const word = useMemo(() => pixelWord(name), [name]);

  // the engine's state lives outside React: nothing re-renders while things move
  const eng = useRef({
    items: new Map<string, Item>(),
    sides: { l: { v: 0, vel: 0, t: 0 }, t: { v: 0, vel: 0, t: 0 }, r: { v: 0, vel: 0, t: 0 }, b: { v: 0, vel: 0, t: 0 } } as Record<"l" | "t" | "r" | "b", Side>,
    handle: null as null | { dir: string; auto: boolean },
    grab: null as null | { id: number; key: string | null; handle: string | null; sx: number; sy: number; ox: number; oy: number; base: Record<"l" | "t" | "r" | "b", number> },
    scale: 1,
    lastTouch: -1e9,
    t0: 0,
    fired: {} as Record<number, number>,
    wake: () => {},
  });

  useEffect(() => {
    const el = root.current;
    const fr = frameEl.current;
    if (!el || !fr) return;
    const E = eng.current;
    const calm = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    E.items.clear();
    el.querySelectorAll<HTMLElement>("[data-drag]").forEach((n) => {
      E.items.set(n.dataset.drag ?? "", { el: n, hang: n.dataset.hang === "1", x: 0, y: 0, vx: 0, vy: 0, th: 0, om: 0, held: 0, tx: 0, ty: 0, rest: true });
    });
    const tags = Array.from(el.querySelectorAll<HTMLElement>("[data-tag]"));
    const smile = el.querySelector<HTMLElement>("[data-smile]");
    const timers: number[] = [];
    let raf = 0;
    let last = 0;
    let seen = true;
    let running = false;

    const showSize = () => {
      if (!readout.current) return;
      const k = (el.offsetWidth > 700 ? 1440 : 390) / Math.max(1, el.offsetWidth);
      readout.current.textContent = `${Math.round(fr.offsetWidth * k)} × ${Math.round(fr.offsetHeight * k)}`;
    };
    showSize();

    const frameLimits = () => ({ w: fr.offsetWidth - E.sides.l.v - E.sides.r.v, h: fr.offsetHeight - E.sides.t.v - E.sides.b.v });

    const tick = (now: number) => {
      raf = 0;
      const dt = Math.min(0.032, last ? (now - last) / 1000 : 0.016);
      last = now;
      const W = el.offsetWidth;
      let busy = false;

      // the unseen hand
      const idle = !calm && now - E.lastTouch > 6000;
      if (idle) {
        if (!E.t0) E.t0 = now;
        const el_t = (now - E.t0) / 1000;
        const cycle = Math.floor(el_t / LOOP);
        const tt = el_t - cycle * LOOP;
        SCRIPT.forEach((s, i) => {
          if ("peel" in s) {
            if (tt >= s.at && E.fired[i] !== cycle) {
              E.fired[i] = cycle;
              tags.forEach((tg, j) => {
                timers.push(window.setTimeout(() => (tg.dataset.peel = "1"), j * 230));
                timers.push(window.setTimeout(() => delete tg.dataset.peel, j * 230 + 820));
              });
            }
            return;
          }
          if ("wink" in s) {
            if (tt >= s.at && E.fired[i] !== cycle && smile) {
              E.fired[i] = cycle;
              smile.dataset.wink = "1";
              timers.push(window.setTimeout(() => delete smile.dataset.wink, 640));
            }
            return;
          }
          const inMove = tt >= s.at && tt < s.at + s.dur + s.hold;
          const p = easeIO(clamp((tt - s.at) / s.dur, 0, 1));
          if ("k" in s) {
            const it = E.items.get(s.k);
            if (!it || it.held === 1) return;
            if (inMove) {
              it.held = 2;
              it.tx = s.to[0] * W * p;
              it.ty = s.to[1] * W * p;
              it.el.dataset.held = "1";
            } else if (it.held === 2) {
              it.held = 0;
              delete it.el.dataset.held;
            }
          } else {
            if (E.handle && !E.handle.auto) return;
            const [hx, hy] = HANDLES[s.handle];
            if (inMove) {
              E.handle = { dir: s.handle, auto: true };
              fr.dataset.pull = s.handle;
              if (hx > 0) E.sides.r.t = s.to[0] * W * p;
              if (hx < 0) E.sides.l.t = -s.to[0] * W * p;
              if (hy > 0) E.sides.b.t = s.to[1] * W * p;
              if (hy < 0) E.sides.t.t = -s.to[1] * W * p;
            } else if (E.handle?.auto && E.handle.dir === s.handle) {
              E.handle = null;
              delete fr.dataset.pull;
            }
          }
        });
        busy = true;
      } else {
        E.t0 = 0;
        E.fired = {};
      }

      // things on the desk
      E.items.forEach((it) => {
        const pvx = it.vx;
        if (it.held) {
          const nx = it.x + (it.tx - it.x) * Math.min(1, dt * 26);
          const ny = it.y + (it.ty - it.y) * Math.min(1, dt * 26);
          it.vx = (nx - it.x) / dt;
          it.vy = (ny - it.y) / dt;
          it.x = nx;
          it.y = ny;
        } else if (calm) {
          it.x = it.y = it.vx = it.vy = it.th = it.om = 0;
        } else {
          it.vx += (-96 * it.x - 11.5 * it.vx) * dt;
          it.vy += (-96 * it.y - 11.5 * it.vy) * dt;
          it.x += it.vx * dt;
          it.y += it.vy * dt;
        }
        if (!calm) {
          if (it.hang) {
            // a pendulum under its tape: the hinge's acceleration swings it
            const ax = clamp((it.vx - pvx) / dt, -9000, 9000);
            it.om += (-46 * it.th - 4.2 * it.om + ax * 0.0042) * dt; // pulled right, the bottom lags left: a clockwise turn
            it.th = clamp(it.th + it.om * dt, -0.6, 0.6);
          } else {
            const want = clamp(it.vx * 0.00022, -0.16, 0.16);
            it.th += (want - it.th) * Math.min(1, dt * 12);
          }
        }
        const still = !it.held && Math.abs(it.x) < 0.05 && Math.abs(it.y) < 0.05 && Math.abs(it.vx) < 1 && Math.abs(it.vy) < 1 && Math.abs(it.th) < 0.0006 && Math.abs(it.om) < 0.004;
        if (still) {
          if (!it.rest) {
            it.rest = true;
            it.x = it.y = it.th = 0;
            it.el.style.transform = "";
          }
        } else {
          it.rest = false;
          busy = true;
          it.el.style.transform = `translate3d(${it.x.toFixed(2)}px, ${it.y.toFixed(2)}px, 0) rotate(calc(var(--r, 0deg) + ${(it.th * 57.3).toFixed(2)}deg))`;
        }
      });

      // the frame's four sides
      const lim = frameLimits();
      let moved = false;
      (["l", "t", "r", "b"] as const).forEach((sd) => {
        const s = E.sides[sd];
        const span = sd === "l" || sd === "r" ? lim.w : lim.h;
        const pulled = E.handle && ((sd === "r" && HANDLES[E.handle.dir][0] > 0) || (sd === "l" && HANDLES[E.handle.dir][0] < 0) || (sd === "b" && HANDLES[E.handle.dir][1] > 0) || (sd === "t" && HANDLES[E.handle.dir][1] < 0));
        if (pulled) {
          const tgt = clamp(s.t, -span * 0.32, span * (sd === "l" || sd === "r" ? 0.5 : 0.7));
          const nv = s.v + (tgt - s.v) * Math.min(1, dt * 24);
          s.vel = (nv - s.v) / dt;
          s.v = nv;
        } else if (calm) {
          s.v = s.vel = 0;
        } else {
          s.vel += (-210 * s.v - 13 * s.vel) * dt;
          s.v += s.vel * dt;
          if (Math.abs(s.v) < 0.05 && Math.abs(s.vel) < 1) s.v = s.vel = 0;
        }
        if (s.v !== 0 || s.vel !== 0 || pulled) moved = true;
        fr.style.setProperty(`--j${sd}`, `${s.v.toFixed(2)}px`);
      });
      if (moved) {
        busy = true;
        showSize();
      }
      if (busy && seen && !document.hidden) raf = requestAnimationFrame(tick);
      else {
        running = false;
        last = 0;
      }
    };
    const wake = () => {
      if (running || !seen || document.hidden) return;
      running = true;
      raf = requestAnimationFrame(tick);
    };
    E.wake = wake;

    const io = new IntersectionObserver(([e]) => {
      seen = e.isIntersecting;
      if (seen) wake();
    });
    io.observe(el);
    const onVis = () => wake();
    document.addEventListener("visibilitychange", onVis);
    const ro = new ResizeObserver(showSize);
    ro.observe(el);
    // the idle hand needs the loop to come back after a touch: check now and then
    const beat = window.setInterval(wake, 1000);
    wake();
    return () => {
      cancelAnimationFrame(raf);
      io.disconnect();
      ro.disconnect();
      document.removeEventListener("visibilitychange", onVis);
      window.clearInterval(beat);
      timers.forEach((t) => window.clearTimeout(t));
    };
  }, [name]);

  /* ---------------------------------------------------------------- the pointer */
  const releaseAuto = () => {
    const E = eng.current;
    E.items.forEach((it) => {
      if (it.held === 2) {
        it.held = 0;
        delete it.el.dataset.held;
      }
    });
    if (E.handle?.auto) {
      E.handle = null;
      if (frameEl.current) delete frameEl.current.dataset.pull;
    }
  };
  const onDown = (e: RPointerEvent<HTMLDivElement>) => {
    const E = eng.current;
    const el = root.current;
    if (!el || E.grab) return;
    const target = e.target as HTMLElement;
    const hEl = target.closest<HTMLElement>("[data-handle]");
    const dEl = target.closest<HTMLElement>("[data-drag]");
    E.lastTouch = performance.now();
    if (!hEl && !dEl) return;
    releaseAuto();
    E.scale = el.getBoundingClientRect().width / Math.max(1, el.offsetWidth) || 1;
    const base = { l: E.sides.l.v, t: E.sides.t.v, r: E.sides.r.v, b: E.sides.b.v };
    if (hEl) {
      const dir = hEl.dataset.handle ?? "e";
      E.grab = { id: e.pointerId, key: null, handle: dir, sx: e.clientX, sy: e.clientY, ox: 0, oy: 0, base };
      E.handle = { dir, auto: false };
      (["l", "t", "r", "b"] as const).forEach((s) => (E.sides[s].t = E.sides[s].v));
      if (frameEl.current) frameEl.current.dataset.pull = dir;
    } else if (dEl) {
      const key = dEl.dataset.drag ?? "";
      const it = E.items.get(key);
      if (!it) return;
      E.grab = { id: e.pointerId, key, handle: null, sx: e.clientX, sy: e.clientY, ox: it.x, oy: it.y, base };
      it.held = 1;
      it.tx = it.x;
      it.ty = it.y;
      it.el.dataset.held = "1";
    }
    try {
      (hEl ?? dEl)?.setPointerCapture(e.pointerId);
    } catch {
      /* fine */
    }
    blip({ f: 2400, to: 900, d: 0.03, g: 0.02, noise: true });
    E.wake();
  };
  const onMove = (e: RPointerEvent<HTMLDivElement>) => {
    const E = eng.current;
    const g = E.grab;
    if (!g || g.id !== e.pointerId) return;
    E.lastTouch = performance.now();
    const dx = (e.clientX - g.sx) / E.scale;
    const dy = (e.clientY - g.sy) / E.scale;
    if (g.handle) {
      const [hx, hy] = HANDLES[g.handle];
      if (hx > 0) E.sides.r.t = g.base.r + dx;
      if (hx < 0) E.sides.l.t = g.base.l - dx;
      if (hy > 0) E.sides.b.t = g.base.b + dy;
      if (hy < 0) E.sides.t.t = g.base.t - dy;
    } else if (g.key) {
      const it = E.items.get(g.key);
      if (it) {
        it.tx = g.ox + dx;
        it.ty = g.oy + dy;
      }
    }
    E.wake();
  };
  const onUp = (e: RPointerEvent<HTMLDivElement>) => {
    const E = eng.current;
    const g = E.grab;
    if (!g || g.id !== e.pointerId) return;
    E.grab = null;
    E.lastTouch = performance.now();
    if (g.handle) {
      E.handle = null;
      if (frameEl.current) delete frameEl.current.dataset.pull;
      blip({ f: 520, to: 300, d: 0.05, g: 0.03 });
    } else if (g.key) {
      const it = E.items.get(g.key);
      if (it) {
        it.held = 0;
        delete it.el.dataset.held;
      }
      blip({ f: 1500, to: 500, d: 0.04, g: 0.024, noise: true });
    }
    E.wake();
  };
  const onKey = (e: KeyboardEvent<HTMLDivElement>) => {
    const d: Record<string, [number, number]> = { ArrowLeft: [-1, 0], ArrowRight: [1, 0], ArrowUp: [0, -1], ArrowDown: [0, 1] };
    const v = d[e.key];
    if (!v) return;
    const E = eng.current;
    const target = e.target as HTMLElement;
    const dEl = target.closest<HTMLElement>("[data-drag]");
    if (!dEl) return;
    e.preventDefault();
    E.lastTouch = performance.now();
    releaseAuto();
    if (dEl.dataset.drag === "frame") {
      // on the frame the arrows stretch the name: a push on a side, which then eases back
      if (v[0] > 0) E.sides.r.vel += 900;
      if (v[0] < 0) E.sides.r.vel -= 700;
      if (v[1] > 0) E.sides.b.vel += 700;
      if (v[1] < 0) E.sides.b.vel -= 500;
      blip({ f: 520, to: 300, d: 0.05, g: 0.03 });
    } else {
      const it = E.items.get(dEl.dataset.drag ?? "");
      if (it) {
        it.vx += v[0] * 700;
        it.vy += v[1] * 700;
      }
      blip({ f: 1500, to: 500, d: 0.04, g: 0.024, noise: true });
    }
    E.wake();
  };

  const tag = (t: (typeof TAGS)[number]) => (
    <div key={t.k} className="jno-tag" data-drag={t.k} data-tag data-k={t.k} data-hand={t.hand} style={{ "--c": t.tone } as CSSProperties} tabIndex={0} role="button" aria-label={`${t.text}. A label you can move with the arrow keys; it springs back.`}>
      <span className="jno-tag-sh" aria-hidden />
      <span className="jno-tag-half jno-tag-flat">
        <span className="jno-tag-txt">{t.text}</span>
      </span>
      <span className="jno-tag-half jno-tag-flap" aria-hidden>
        <span className="jno-tag-txt">{t.text}</span>
      </span>
    </div>
  );
  const polaroid = (k: string, src: string, cap: string, tape: string) => (
    <div key={k} className="jno-pol" data-drag={k} data-hang="1" data-k={k} tabIndex={0} role="button" aria-label={`Photograph: ${cap}. Arrow keys nudge it; it swings back.`}>
      <span className="jno-tape" style={{ "--c": tape } as CSSProperties} aria-hidden />
      <span className="jno-pol-pic">
        {/* eslint-disable-next-line @next/next/no-img-element */}
        <img src={src} alt="" draggable={false} />
      </span>
      <span className="jno-pol-cap">{cap}</span>
    </div>
  );

  return (
    <div ref={root} className="jno-root" data-paper={paper} style={{ "--jno-frame": frame } as CSSProperties} onPointerDown={onDown} onPointerMove={onMove} onPointerUp={onUp} onPointerCancel={onUp} onKeyDown={onKey}>
      <style>{css}</style>
      <span className="jno-paper" aria-hidden />
      <span className="jno-margin" aria-hidden />

      <nav className="jno-nav" aria-label="Juno">
        <span className="jno-dot" aria-hidden>
          <i />
        </span>
        <ul className="jno-tabs">
          {TABS.map((t, i) => (
            <li key={t} data-on={i === 0} data-i={i}>
              <span className="jno-ico" data-i={i} aria-hidden />
              {t}
            </li>
          ))}
        </ul>
        <span className="jno-nav-r">
          <i className="jno-round" data-c="y" aria-hidden />
          <i className="jno-round" data-c="p" aria-hidden />
          <i className="jno-round" data-c="g" aria-hidden />
          <span className="jno-contact">
            <i aria-hidden /> Contact
          </span>
        </span>
      </nav>

      <p className="jno-hello">
        my name is
        <svg viewBox="0 0 80 10" aria-hidden>
          <path d="M3 5.5c16-4 38-4.6 74-1.2" />
        </svg>
      </p>

      {TAGS.map(tag)}
      <svg className="jno-arrow" data-k="a1" viewBox="0 0 60 40" aria-hidden>
        <path d="M4 5c18 1 34 9 47 27" />
        <path d="M41 28l10 5 1-12" />
      </svg>
      <svg className="jno-arrow" data-k="a2" viewBox="0 0 60 40" aria-hidden>
        <path d="M56 6C38 7 22 15 9 31" />
        <path d="M19 28L8 32 8 20" />
      </svg>
      <svg className="jno-arrow" data-k="a3" viewBox="0 0 60 40" aria-hidden>
        <path d="M5 34c14-2 30-9 46-26" />
        <path d="M40 9l12-2-3 12" />
      </svg>

      <div ref={frameEl} className="jno-frame" data-drag="frame" tabIndex={0} role="button" aria-label={`${name}, in a selection frame. Arrow keys stretch the name; it eases back.`}>
        <h1 className="jno-name" aria-label={name}>
          <svg viewBox={`0 0 ${word.w} 9`} preserveAspectRatio="none" shapeRendering="crispEdges" aria-hidden>
            {word.runs.map((r, i) => (
              <rect key={i} x={r.x} y={r.y} width={r.w} height={1.03} />

          </svg>

        {Object.keys(HANDLES).map((h) => (
          <i key={h} className="jno-h" data-handle={h} aria-hidden />
        ))}
        <span className="jno-size" aria-hidden>
          <span ref={readout}>389 × 150</span>
        </span>
      </div>

      <p className="jno-open">
        <i aria-hidden /> Open to new work from January

      {polaroid("p1", portrait, "that's me", "rgba(157,184,242,.78)")}
      {polaroid("p2", street, "Tramline, 2026", "rgba(246,214,103,.8)")}

        <div key={n.k} className="jno-note" data-drag={n.k} data-k={n.k} style={{ "--c": n.tone } as CSSProperties} tabIndex={0} role="button" aria-label={`${n.big} ${n.small}, ${n.by}. A note you can move with the arrow keys.`}>

            className="jno-smile-bg"
            d="M20 1.5l3.6 3.1 4.6-1.1 1.9 4.3 4.6.9-.3 4.7 3.9 2.6-2.3 4.1 2.3 4.1-3.9 2.6.3 4.7-4.6.9-1.9 4.3-4.6-1.1L20 38.500l-3.6-3.100-4.6 1.100-1.900-4.300-4.600-.900.300-4.700-3.900-2.600 2.300-4.100-2.300-4.100 3.900-2.600-.300-4.700 4.600-.900 1.900-4.300 4.600 1.100z"

The full brief of this design, for reference, or to rebuild it in another stack:

# Juno — a designer's portfolio as a page of her notebook (hero, one screen)

Build the first screen of a product designer's portfolio. It is a sheet of pale lined paper with things stuck on it, and the whole point is that you can pick those things up. No scrolling, no sections: one screen, 1440×900 on desktop, recomposed at 390×844.

## Pictures (three square photographs, shown as polaroids)
Generate them as real, candid 35mm snapshots with soft grain, no readable text, no logos:
1. `juno.webp` — "a woman product designer in her early thirties with short dark curly hair, mustard yellow beanie, dark green overshirt, seen in three-quarter view from behind so only her cheek shows, standing in a bright studio looking at a white wall covered with paper sketches of phone screens and pastel sticky notes with no writing, soft daylight from a window on the left, shallow depth of field".
2. `tram.webp` — "close view of a hand holding a smartphone at a tram stop in Lisbon on a bright day, the screen shows a pale map with one bold red route line and a round marker, a yellow tram passing out of focus behind, pastel tiled buildings".
3. `desk.webp` — "a designer's wooden desk by a window in warm late-afternoon light, a smartphone showing a colourful map with one route line lying on pencil wireframe sketches, a pencil, blank pastel sticky notes, a ceramic coffee cup, a roll of masking tape, three-quarter angle from above".
Export at 640 px wide.

## Type
- Pixel face for small labels: **Silkscreen** 400/700 (Google Fonts), uppercase, letter-spacing .04–.08em, 10–11 px at 1440.
- Handwriting: **Reenie Beanie** (Google Fonts), 23–30 px at 1440.
- Statement: **Inter Tight** 600, 37 px at 1440, tracking −0.022em, line-height 1.16.
- The NAME is not a font. Draw it from a pixel alphabet you define yourself: each letter is 9 rows by 6 columns (M and W 8, N 7), strokes two cells thick, corners notched by one cell, one empty column between letters. Render the lit cells as SVG rects in an `<svg viewBox="0 0 cols 9" preserveAspectRatio="none" shape-rendering="crispEdges">` so the word can be squashed and stretched and stays sharp.

## Colours
Paper `#f1f2ee`; rules `rgba(64,92,140,.13)`, one every 27 px; a faint red margin line `rgba(240,69,46,.16)` at 75 px from the left. Ink `#17181a`. Frame red `#f0452e`. Yellow `#f6d048` / tape yellow `#f6d667`. Mint `#a9e5c5`. Pink `#ff7eb0`. Blue `#2f6bff` (button icon) and `#b9cdf6` (note). Green `#37c47c`. Polaroid white `#fdfdfb`.

## Motion (one idea: the page is a desk you can touch)
- **Pick up anything** that is taped or framed (labels, polaroids, notes, sticker, the frame itself). While held it follows the pointer, casts a deeper, softer shadow and tilts with its speed (up to ±9°). On release a spring takes it home (stiffness 96, damping 11.5: one small overshoot).
- **Polaroids hang from their tape**: `transform-origin: 50% 1%`, and the angle is a pendulum driven by the hinge's horizontal acceleration: `ω += (−46·θ − 4.2·ω + a·0.0042)·dt`. They swing behind the hand and settle in about a second.
- **The frame's handles** resize it like a design tool: each of the four sides is its own spring; a handle pulls one or two sides, the name stretches with the frame, the readout counts. On release the sides return (stiffness 210, damping 13, a visible bounce). Limits: −32% to +50% wide, +70% tall.
- **Tape labels peel** on hover and focus: the label is two clipped copies of itself, the free 62% turns around the fold with `rotateY(−52deg) rotateZ(−3deg)` under a parent `perspective`, brightens 8% and uncovers a blurred shadow; 500 ms, `cubic-bezier(.3,1.5,.5,1)`.
- **Self-play** when nobody has touched it for 6 s, in a 14 s loop, silent: 0.7 s a polaroid is pulled 144 px down-left and let go; 3.1 s the frame's bottom-right handle is pulled out and let go; 5.5 s the labels peel one after another, 230 ms apart; 7.5 s the smiley winks and hops; 8.3 s the left polaroid is pulled up-right; 10.3 s a note; 11.7 s the frame's left handle.
- Everything stops off screen and in a hidden tab. Reduced motion: no self-play, no peel, no bounce (a dropped thing is simply back).
- Sound: a short noise tick on pick-up, a softer one on release, a low tap when the frame lets go; each under 60 ms and quiet, never before a user gesture.
