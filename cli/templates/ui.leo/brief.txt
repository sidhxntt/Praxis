```tsx
"use client";

/**
 * LEO · website hero
 *
 * The first screen of a film photographer's portfolio, Leo Marchetti. His
 * name is set huge in signal red on a pale wall, and he stands in front of it,
 * a black-and-white cut-out holding his camera. The idea is the glance: Leo
 * turns his head to look at you. Move the pointer and he follows it, left,
 * centre, right; leave him alone and he glances slowly around the room on his
 * own. The turn is real, not a skew: it plays a set of photographs of the
 * same man at different head angles, with the in-between frames morphed by
 * optical flow, so his nose, beard and curls rotate like a person's.
 *
 * Three planes part in a slow dolly and a little more under the pointer: the
 * red name at the back (with Leo's soft shadow on it), Leo, and his film
 * prints at the bottom right in front. The prints shuffle on a rhythm, the
 * frame counter advances with each one; hover a print to lift it, press it to
 * bring it to the top (a soft shutter click).
 *
 * One file, React only. The typeface (Archivo, variable width) and the mono
 * (JetBrains Mono) come from Google Fonts, loaded in the CSS.
 *
 *   <Leo />
 *   <Leo ink="#1f3fd6" name="Leo" />
 *
 * Props
 *   ink       the colour of the giant name               "#e8321e"
 *   ground    the wall                                   "#f2f1ee"
 *   name      the first name in "HI, I'M …"              "Leo"
 *   fullName  the wordmark                               "Leo Marchetti"
 *   line      the one line about him                     (see default)
 *   cta       the main button                            "Commission"
 *   body      cut-out of the body, head zone erased      "/design/websites/leo/body.webp"
 *   heads     sprite of the head-turn frames, left→right "/design/websites/leo/heads.webp"
 *   prints    sprite of four prints, 2 × 2               "/design/websites/leo/prints.webp"
 *   roll      the roll number on the counter             214
 *   sound     tiny synthesized sounds on interaction     true
 *   onCta     called when the main button is pressed
 *
 * Fills its container's width; height is var(--design-h, 100svh), never under
 * 640px. Responsive by container query: on a phone the name stacks ("HI, I'M"
 * over a huge "LEO"), Leo stands centred in front of it and the prints sit at
 * the bottom right. The loop pauses off screen and in a hidden tab.
 * Keyboard: Leo is focusable, left/right arrows turn his head; each print is a
 * button (Enter/Space brings it to the top). Reduced motion: no entrance, no
 * drift, no shuffle, he looks at the camera (and still turns to the arrows).
 * Sound: silent until the visitor interacts, when `sound` is false, and when
 * <html data-cd-mute="1">. The AudioContext is created on the first sound.
 */

import { useCallback, useEffect, useRef, useState, type CSSProperties, type KeyboardEvent } from "react";

type Props = {
  ink?: string;
  ground?: string;
  name?: string;
  fullName?: string;
  line?: string;
  cta?: string;
  body?: string;
  heads?: string;
  prints?: string;
  roll?: number;
  sound?: boolean;
  onCta?: () => void;
};

/* The head-turn sprite: FRAMES frames, left (his gaze to the left of the frame)
   to right, COLS per row. Each frame covers HEAD (fractions of the body image). */
const FRAMES = 25;
const COLS = 5;
const ROWS = Math.ceil(FRAMES / COLS);
const HEAD = { x: 0.1953, y: 0, w: 0.6152, h: 0.4167 };
/* Where his face is, across the body image (for the pointer). */
const FACE_X = 0.48;
/* He rests only on the five photographed angles (far left … far right);
   the morphed frames between them are only ever seen in motion. */
const KEYS = [-1, -0.5, 0, 0.5, 1];
/* His own glance when nobody points: [angle, seconds he holds it]. */
const GLANCE: [number, number][] = [
  [0, 1.0],
  [1, 1.5],
  [0, 0.9],
  [-1, 1.5],
];
const MOVE = 0.95; // seconds a glance takes to arrive

/* The four prints and where they lie (px at 1440 wide; scaled with the hero). */
const PRINT_ALT = ["Procida harbour at sunset", "A night train on a viaduct in the Alps", "A woman among peonies", "A portrait on mint green"];
const SLOTS = [
  { x: 172, y: 26, r: 5 },
  { x: 0, y: 64, r: -6 },
  { x: 96, y: 8, r: -1.5 },
  { x: 130, y: 0, r: 2.5 },
];
const SHUFFLE_EVERY = 3.2; // seconds between shuffles
const SHUFFLE_TIME = 1.15; // seconds a shuffle takes

const ease = (t: number) => (t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2);
const clamp = (v: number, a: number, b: number) => Math.min(b, Math.max(a, v));

export default function Leo({
  ink = "#e8321e",
  ground = "#f2f1ee",
  name = "Leo",
  fullName = "Leo Marchetti",
  line = "I shoot people and places on 35 mm film. Ten years, forty countries, one camera.",
  cta = "Commission",
  body = "/design/websites/leo/body.webp",
  heads = "/design/websites/leo/heads.webp",
  prints = "/design/websites/leo/prints.webp",
  roll = 214,
  sound = true,
  onCta,
}: Props) {
  const root = useRef<HTMLElement>(null);
  const typeRef = useRef<HTMLDivElement>(null);
  const manRef = useRef<HTMLDivElement>(null);
  const shadowRef = useRef<HTMLDivElement>(null);
  const deckRef = useRef<HTMLDivElement>(null);
  const headA = useRef<HTMLDivElement>(null);
  const headB = useRef<HTMLDivElement>(null);
  const cards = useRef<(HTMLButtonElement | null)[]>([]);
  const audio = useRef<AudioContext | null>(null);
  const [frame, setFrame] = useState(17);
  const [ready, setReady] = useState(false);
  const [busy, setBusy] = useState(false);

  /* Shared state the loop reads (no re-renders per frame). */
  const st = useRef({
    order: [0, 1, 2, 3] as number[], // order[slot] = print index; slot 0 is on top
    shuffleAt: 0, // time the current shuffle started (s), -1 = none
    from: [0, 1, 2, 3] as number[], // slot each print starts from in the shuffle
    flyer: -1, // print that leaves the top and goes to the back
    hover: -1,
    lift: [0, 0, 0, 0],
    nextShuffle: SHUFFLE_EVERY,
    px: 0,
    py: 0,
    pActive: 0, // time of the last pointer move (s)
    keyTurn: 0, // -1..1 from the arrow keys
    keyAt: -99,
    angle: 0,
    key: 2,
    vel: 0,
    cam: 0,
    camY: 0,
    reduced: false,
  });

  /* ── sound ───────────────────────────────────────────────────────────── */
  const ac = useCallback((): AudioContext | null => {
    if (!sound || document.documentElement.dataset.cdMute === "1") return null;
    try {
      if (!audio.current) {
        const AC = window.AudioContext || (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
        if (!AC) return null;
        audio.current = new AC();
      }
      if (audio.current.state === "suspended") void audio.current.resume();
      return audio.current;
    } catch {
      return null;
    }
  }, [sound]);

  /* A film-camera shutter: a short high click, 30 ms of filtered noise (the
     curtain), and a lower click 40 ms later (the mirror). Peak .05. */
  const shutter = useCallback(() => {
    const a = ac();
    if (!a) return;
    try {
      const t = a.currentTime;
      const len = Math.floor(a.sampleRate * 0.035);
      const buf = a.createBuffer(1, len, a.sampleRate);
      const d = buf.getChannelData(0);
      for (let i = 0; i < len; i++) d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / len, 2);
      const n = a.createBufferSource();
      n.buffer = buf;
      const bp = a.createBiquadFilter();
      bp.type = "bandpass";
      bp.frequency.value = 3200;
      bp.Q.value = 0.8;
      const g = a.createGain();
      g.gain.value = 0.05;
      n.connect(bp).connect(g).connect(a.destination);
      n.start(t);
      [0, 0.045].forEach((dt, k) => {
        const o = a.createOscillator();
        const og = a.createGain();
        o.type = "square";
        o.frequency.setValueAtTime(k ? 900 : 2100, t + dt);
        og.gain.setValueAtTime(0.0001, t + dt);
        og.gain.exponentialRampToValueAtTime(k ? 0.03 : 0.04, t + dt + 0.002);
        og.gain.exponentialRampToValueAtTime(0.0001, t + dt + 0.018);
        o.connect(og).connect(a.destination);
        o.start(t + dt);
        o.stop(t + dt + 0.03);
      });
    } catch {
      /* no sound, no problem */
    }
  }, [ac]);

  /* A soft tick for the button. */
  const tick = useCallback(() => {
    const a = ac();
    if (!a) return;
    try {
      const t = a.currentTime;
      const o = a.createOscillator();
      const g = a.createGain();
      o.type = "sine";
      o.frequency.setValueAtTime(1250, t);
      o.frequency.exponentialRampToValueAtTime(700, t + 0.04);
      g.gain.setValueAtTime(0.0001, t);
      g.gain.exponentialRampToValueAtTime(0.05, t + 0.004);
      g.gain.exponentialRampToValueAtTime(0.0001, t + 0.05);
      o.connect(g).connect(a.destination);
      o.start(t);
      o.stop(t + 0.06);
    } catch {
      /* ignore */
    }
  }, [ac]);

  /* ── prints: start a shuffle that brings print `p` to the top ───────────── */
  const clock = useRef(0);
  const shuffleTo = useCallback((p: number | null) => {
    const s = st.current;
    if (s.shuffleAt >= 0) return false;
    const cur = s.order.slice();
    let next: number[];
    let flyer: number;
    if (p === null) {
      flyer = cur[0];
      next = [cur[1], cur[2], cur[3], cur[0]];
    } else {
      if (cur[0] === p) return false;
      flyer = p;
      next = [p, ...cur.filter((x) => x !== p)];
    }
    s.from = [0, 1, 2, 3].map((i) => cur.indexOf(i));
    s.order = next;
    s.flyer = flyer;
    s.shuffleAt = s.reduced ? -1 : clock.current;
    setFrame((f) => (f >= 36 ? 1 : f + 1));
    return true;
  }, []);

  /* ── the loop ────────────────────────────────────────────────────────── */
  useEffect(() => {
    const el = root.current;
    if (!el) return;
    const s = st.current;
    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
    s.reduced = mq.matches;
    const onMq = () => (s.reduced = mq.matches);
    mq.addEventListener?.("change", onMq);

    let raf = 0;
    let visible = true;
    let last = performance.now();
    let t = 0;
    let shownA = -1;
    let shownB = -1;
    let opB = -1;
    // Two neighbouring frames are mixed exactly (A × (1 − f) + B × f, edges included) where the
    // browser can add layers; elsewhere B simply fades in over A.
    const exact = typeof CSS !== "undefined" && CSS.supports("mix-blend-mode", "plus-lighter");

    const place = (node: HTMLDivElement | null, f: number) => {
      if (!node) return;
      const c = f % COLS;
      const r = Math.floor(f / COLS);
      node.style.backgroundPosition = `${(c / (COLS - 1)) * 100}% ${(r / Math.max(1, ROWS - 1)) * 100}%`;
    };

    const step = (now: number) => {
      const dt = Math.min(0.05, (now - last) / 1000);
      last = now;
      t += dt;
      clock.current = t;
      const W = el.clientWidth || 1440;
      const unit = W / 1440;
      const phone = W < 640;
      const pu = phone ? unit * 2.3 : unit; // the prints are bigger on a phone

      /* where he looks: pointer, arrows, or his own slow glance */
      let target = 0;
      const pointerOn = t - s.pActive < 2.6 && s.pActive > 0;
      if (t - s.keyAt < 4) target = s.keyTurn;
      else if (pointerOn) {
        const man = manRef.current;
        const r = man ? man.getBoundingClientRect() : null;
        const box = el.getBoundingClientRect();
        const cx = r ? r.left - box.left + r.width * FACE_X : W / 2;
        const raw = clamp((s.px - cx) / (W * (phone ? 0.36 : 0.26)), -1, 1);
        // snap to the nearest photographed angle, with a little hysteresis
        const cur = KEYS[s.key];
        if (Math.abs(raw - cur) > 0.25 + 0.06) {
          let best = 0;
          for (let i = 1; i < KEYS.length; i++) if (Math.abs(KEYS[i] - raw) < Math.abs(KEYS[best] - raw)) best = i;
          s.key = best;
        }
        target = KEYS[s.key];
      } else if (!s.reduced) {
        let span = 0;
        for (const g of GLANCE) span += g[1] + MOVE;
        let u = (t + 0.4) % span;
        let i = 0;
        while (u > GLANCE[i][1] + MOVE) {
          u -= GLANCE[i][1] + MOVE;
          i++;
        }
        target = GLANCE[i][0];
      }
      if (!pointerOn) s.key = KEYS.indexOf(target) >= 0 ? KEYS.indexOf(target) : 2;
      // critically damped: eases in and out like a neck, never overshoots
      const k = pointerOn || t - s.keyAt < 4 ? 30 : 17;
      const acc = (target - s.angle) * k - 2 * Math.sqrt(k) * s.vel;
      s.vel += acc * dt;
      s.angle += s.vel * dt;
      const a = clamp(s.angle, -1, 1);
      const f = ((a + 1) / 2) * (FRAMES - 1);
      const i0 = Math.floor(f);
      const i1 = Math.min(FRAMES - 1, i0 + 1);
      const fr = f - i0;
      if (i0 !== shownA) {
        place(headA.current, i0);
        shownA = i0;
      }
      if (i1 !== shownB) {
        place(headB.current, i1);
        shownB = i1;
      }
      const ob = Math.round(fr * 100) / 100;
      if (ob !== opB && headB.current) {
        headB.current.style.opacity = String(ob);
        if (exact && headA.current) headA.current.style.opacity = String(1 - ob);
        opB = ob;
      }

      /* the dolly: slow on its own, more under the pointer */
      const amb = s.reduced ? 0 : Math.sin((t / 15) * Math.PI * 2);
      const ambY = s.reduced ? 0 : Math.sin((t / 11) * Math.PI * 2 + 1.1);
      const tx = pointerOn ? clamp((s.px / W) * 2 - 1, -1, 1) : 0;
      const ty = pointerOn ? clamp((s.py / (el.clientHeight || 810)) * 2 - 1, -1, 1) : 0;
      const kc = 1 - Math.exp(-dt * 3);
      s.cam += (amb * 0.75 + tx * 0.9 - s.cam) * kc;
      s.camY += (ambY * 0.5 + ty * 0.7 - s.camY) * kc;
      const cx = s.cam * unit;
      const cy = s.camY * unit;
      if (typeRef.current) typeRef.current.style.transform = `translate3d(${(cx * 9).toFixed(2)}px, ${(cy * 4).toFixed(2)}px, 0)`;
      if (manRef.current) manRef.current.style.transform = `translate3d(${(-cx * 10).toFixed(2)}px, ${(-cy * 5).toFixed(2)}px, 0)`;
      if (shadowRef.current) shadowRef.current.style.transform = `translate3d(${(-cx * 3 + 34 * unit).toFixed(2)}px, ${(-cy * 1 + 16 * unit).toFixed(2)}px, 0)`;
      if (deckRef.current) deckRef.current.style.transform = `translate3d(${(-cx * 26).toFixed(2)}px, ${(-cy * 12).toFixed(2)}px, 0)`;

      /* prints */
      if (!s.reduced && s.hover < 0 && s.shuffleAt < 0 && t >= s.nextShuffle) {
        shuffleTo(null);
      }
      let p = 1;
      if (s.shuffleAt >= 0) {
        p = clamp((t - s.shuffleAt) / SHUFFLE_TIME, 0, 1);
        if (p >= 1) {
          s.shuffleAt = -1;
          s.flyer = -1;
          s.nextShuffle = t + SHUFFLE_EVERY - SHUFFLE_TIME;
        }
      }
      const e = ease(p);
      for (let i = 0; i < 4; i++) {
        const c = cards.current[i];
        if (!c) continue;
        const to = s.order.indexOf(i);
        const fromSlot = s.shuffleAt >= 0 ? s.from[i] : to;
        const A = SLOTS[fromSlot];
        const B = SLOTS[to];
        let x = A.x + (B.x - A.x) * e;
        let y = A.y + (B.y - A.y) * e;
        let r = A.r + (B.r - A.r) * e;
        let z = 10 - to;
        if (s.shuffleAt >= 0) {
          const arc = Math.sin(Math.PI * e);
          if (i === s.flyer && to > fromSlot) {
            // leaves the top: lifts clear, swings right, slides in at the back
            y -= arc * 168;
            x += arc * 70;
            r += arc * 7;
            z = e < 0.5 ? 20 : 1;
          } else if (i === s.flyer) {
            // comes up from the back: rises clear, then lands on top
            y -= arc * 168;
            x -= arc * 40;
            r -= arc * 5;
            z = e < 0.5 ? 1 : 20;
          } else z = 10 - (e < 0.5 ? fromSlot : to);
        }
        const want = s.hover === i && s.shuffleAt < 0 ? 1 : 0;
        s.lift[i] += (want - s.lift[i]) * (1 - Math.exp(-dt * 14));
        const L = s.lift[i];
        y -= L * 16;
        r *= 1 - L * 0.5;
        const sc = 1 + L * 0.06;
        c.style.transform = `translate3d(${(x * pu).toFixed(2)}px, ${(y * pu).toFixed(2)}px, 0) rotate(${r.toFixed(2)}deg) scale(${sc.toFixed(3)})`;
        c.style.zIndex = String(L > 0.05 ? 30 : z);
        c.style.setProperty("--leo-lift", L.toFixed(3));
      }

      raf = visible && !document.hidden ? requestAnimationFrame(step) : 0;
    };

    const start = () => {
      if (!raf && visible && !document.hidden) {
        last = performance.now();
        raf = requestAnimationFrame(step);
      }
    };
    const io = new IntersectionObserver(([en]) => {
      visible = en.isIntersecting;
      if (visible) start();
      else if (raf) {
        cancelAnimationFrame(raf);
        raf = 0;
      }
    });
    io.observe(el);
    const onVis = () => {
      if (!document.hidden) start();
    };
    document.addEventListener("visibilitychange", onVis);

    const onMove = (ev: PointerEvent) => {
      const b = el.getBoundingClientRect();
      s.px = ev.clientX - b.left;
      s.py = ev.clientY - b.top;
      s.pActive = clock.current || 0.001;
    };
    const onLeave = () => {
      s.pActive = 0;
    };
    el.addEventListener("pointermove", onMove);
    el.addEventListener("pointerleave", onLeave);

    start();
    const id = window.setTimeout(() => setReady(true), 30);
    return () => {
      cancelAnimationFrame(raf);
      io.disconnect();
      document.removeEventListener("visibilitychange", onVis);
      el.removeEventListener("pointermove", onMove);
      el.removeEventListener("pointerleave", onLeave);
      mq.removeEventListener?.("change", onMq);
      window.clearTimeout(id);
    };
  }, [shuffleTo]);

  useEffect(
    () => () => {
      void audio.current?.close().catch(() => {});
    },
    [],
  );

  const pick = (i: number) => {
    if (shuffleTo(i)) shutter();
    else if (st.current.order[0] === i) shutter();
  };

  const onManKey = (e: KeyboardEvent<HTMLDivElement>) => {
    const s = st.current;
    if (e.key === "ArrowLeft" || e.key === "ArrowRight") {
      e.preventDefault();
      s.keyTurn = clamp(s.keyTurn + (e.key === "ArrowLeft" ? -0.5 : 0.5), -1, 1);
      s.keyAt = clock.current;
      tick();
    } else if (e.key === "Home") {
      s.keyTurn = 0;
      s.keyAt = clock.current;
    }
  };

  const press = () => {
    tick();
    setBusy(true);
    onCta?.();
    // Demo only: the button resets after 1.8 s so the next visitor can press it.
    window.setTimeout(() => setBusy(false), 1800);
  };

  const vars = { "--leo-ink": ink, "--leo-ground": ground } as CSSProperties;
  const sprite = (src: string): CSSProperties => ({ backgroundImage: `url("${src}")`, backgroundSize: `${COLS * 100}% ${ROWS * 100}%` });
  const centre = Math.floor(FRAMES / 2);
  const centrePos = `${((centre % COLS) / (COLS - 1)) * 100}% ${(Math.floor(centre / COLS) / Math.max(1, ROWS - 1)) * 100}%`;
  const headBox: CSSProperties = { left: `${HEAD.x * 100}%`, top: `${HEAD.y * 100}%`, width: `${HEAD.w * 100}%`, height: `${HEAD.h * 100}%` };

  return (
    <section ref={root} className={`leo-root${ready ? " leo-in" : ""}`} style={vars} aria-label={`${fullName}, film photographer`}>
      <style>{css}</style>
      <div className="leo-wall" aria-hidden="true" />

      <nav className="leo-nav">
        <span className="leo-logo">{fullName}</span>
        <div className="leo-links">
          <a href="#work">Work</a>
          <a href="#prints">Prints</a>
          <a href="#workshops">Workshops</a>
        </div>
        <button type="button" className="leo-cta" onClick={press} aria-live="polite">
          {busy ? "Thanks, I'll write back" : cta}
        </button>
      </nav>

      <div className="leo-type" ref={typeRef}>
        <h1 className="leo-h1">
          <span className="leo-w leo-w1">
            <span>HI,</span>
          </span>{" "}
          <span className="leo-w leo-w2">
            <span>I&rsquo;M</span>
          </span>{" "}
          <span className="leo-w leo-w3">
            <span>{name.toUpperCase()}</span>
          </span>
        </h1>
      </div>

      <div className="leo-figure leo-shadowbox" ref={shadowRef} aria-hidden="true">
        <img className="leo-body" src={body} alt="" draggable={false} />
        <div className="leo-head" style={{ ...headBox, ...sprite(heads), backgroundPosition: centrePos }} />
      </div>

      <div
        className="leo-figure leo-man"
        ref={manRef}
        tabIndex={0}
        role="img"
        aria-label={`${fullName} holding a 35 mm camera. He turns his head toward the pointer; use the left and right arrow keys to turn it.`}
        onKeyDown={onManKey}
      >
        <img className="leo-body" src={body} alt="" draggable={false} />
        <div className="leo-heads" style={headBox}>
          <div className="leo-head leo-head-a" ref={headA} style={{ ...sprite(heads), backgroundPosition: centrePos }} />
          <div className="leo-head leo-head-b" ref={headB} style={{ ...sprite(heads), backgroundPosition: centrePos, opacity: 0 }} />
        </div>
      </div>

      <p className="leo-line">{line}</p>

      <div className="leo-deck" ref={deckRef} id="prints" aria-label="Prints">
        {[0, 1, 2, 3].map((i) => (
          <button
            key={i}
            type="button"
            className="leo-print"
            ref={(n) => {
              cards.current[i] = n;
            }}
            style={{ backgroundImage: `url("${prints}")`, backgroundPosition: `${(i % 2) * 100}% ${Math.floor(i / 2) * 100}%` }}
```

The full brief of this design, for reference, or to rebuild it in another stack:

# Leo · film photographer hero

The first screen of a film photographer's portfolio, Leo Marchetti. A pale wall (#f2f1ee with a faint paper grain), his greeting set enormous in signal red, "HI, I'M LEO", and Leo himself in front of it: a black-and-white waist-up cut-out of a bearded man in a worn canvas work jacket over a flannel shirt, holding a 35 mm SLR at his chest. The idea is the glance. Leo turns his head to look at the visitor: the pointer moves and he follows it (far left, left, centre, right, far right); left alone he glances slowly around the room on his own. The turn must look like a real person turning, never a skew: it is a set of real photographs of the same man at five head angles, and the in-between frames are morphed with optical flow, so the nose, beard and curls rotate.

## Images (three files)

1. **The five angles.** Generate the centre photo first (portrait 2:3, gpt-image class): "Black and white documentary-style photograph of a bearded man in his late twenties holding a vintage 35mm film camera at chest height, wearing a canvas work jacket, slight smile, looking at the viewer, natural light, real film grain, plain light grey background, framed from the waist up." Then make four image-to-image EDITS of that photo (feed it in as the input image; do not generate from scratch): "the man turns his head about 15° / 30° toward the left / right side of the frame. Only the head and neck rotate, his eyes lead the turn. Everything else stays pixel-identical: same face identity, beard, curls, mole, expression, jacket, strap, camera and hands, same light, same grey background, same crop." Check identity across the five.
2. **body.webp** (1024×1536, alpha): the centre photo cut out (Vision "lift subject"), edges refined with a luminance key against the grey and the grey unmixed out of the soft edge (no light halo on red), with the head zone erased above the collar.
3. **heads.webp** (sprite 5 × 5, 25 frames of 504×512, alpha): the head + neck crop (x 200–830, y 0–640 of the 1024×1536 canvas) for 25 angles, far left → far right. Frames 0, 6, 12, 18, 24 are the five photographs (registered to the body with ORB + RANSAC; all greyscale, tone-matched). The 5 frames between each pair are a turn, not a cross-dissolve: an affine fit of one head onto the other, DIS optical flow (medium preset, both directions) on the residual, then the first photograph alone warped along the flow for the first two frames, the second alone for the last two, and one 50/50 frame in the middle, so every frame is sharp. Edges are cleaned (largest shape kept, alpha smoothed). Each frame fades into the body across the collar (rows 505–600 at the neck, 430–500 at the shoulders) so the seam is invisible. On the page two neighbouring frames are shown at once and added exactly (A × (1 − f) + B × f, `mix-blend-mode: plus-lighter` in an isolated group), so the outline never pops.
4. **prints.webp** (1440×960, 2 × 2 of 720×480): four colour film prints. Procida harbour at sunset (Portra 400, Leica M6 35 mm), a red night train on an Alpine viaduct, a freckled woman behind pink peonies in sun, a woman in a white shirt on mint green.

## Layers (back to front)

1. Wall #f2f1ee + a soft white light top-left + paper grain (SVG fractal noise, multiply 10%).
2. **The name** "HI, I'M LEO": Archivo 900, width 62 (variable `wdth`), 20.83cqi (300 px at 1440), line-height .82, tracking −0.01em, #e8321e, centred, top 14.8%.
3. **Leo's shadow** on the wall and letters: the body + centre head, `filter: brightness(0) blur(12px)`, opacity .22, offset 34 / 16 px (scaled), moving less than Leo so it slides across the letters with the dolly.
4. **Leo**: 2:3 box, height 86.4% of the hero, centre at 46.1cqi, bottom −4.9%. The body image, then two head layers from the sprite crossfading between neighbouring frames.
5. **The line** bottom-left (Inter Tight 16/1.45, max 330 px): "I shoot people and places on 35 mm film. Ten years, forty countries, one camera."
6. **The prints** bottom-right (13.2cqi × 9.03cqi each, 6 px white border, two-layer shadow), four fanned slots: (172, 26, 5°) on top, (0, 64, −6°), (96, 8, −1.5°), (130, 0, 2.5°).
7. "ROLL 214 · FRAME 17", JetBrains Mono 12 px, 0.1em, bottom-right; the frame advances with every shuffle.
8. Nav: "Leo Marchetti" (Inter Tight 600 18 px) · Work, Prints, Workshops (14 px 500, 80%) · black pill "Commission".

## Motion

- **The head** (the signature): an angle −1…1, driven by a critically damped spring (k 30 under the pointer, 17 on its own), so it eases in and out like a neck and never overshoots. The angle picks the two nearest frames and crossfades them. Under the pointer the target is the pointer's x relative to his face, snapped to the five photographed angles with hysteresis, so he only ever rests on a real photograph; the morph frames are seen only in motion. With no pointer he glances: centre 1.0 s → far right 1.5 s → centre 0.9 s → far left 1.5 s, each move ~0.95 s. Arrow keys turn him (Home recentres).
- **The dolly**: the name, Leo, his shadow and the prints drift apart on slow sines (15 s horizontal, 11 s vertical) and a little more under the pointer: name +9 px, Leo −10 px, shadow −3 px, prints −26 px per unit.
- **The shuffle**: every 3.2 s the top print lifts clear (168 px up, 70 px right, +7°), swings over and slides in at the back while the others advance one slot (1.15 s, cubic in-out). Hover a print: the shuffle waits, the print lifts 16 px and scales 1.06 with a deeper shadow. Press it: it rises from the pile and lands on top with a soft synthesized shutter (noise curtain + two clicks, peak .05).
- **Entrance** (done by 1.5 s): the three words rise out of masks (1 s, 80 ms stagger), Leo fades in from 1.025 scale, the line and prints follow.
- Paused off screen and in hidden tabs. Reduced motion: no entrance, no drift, no shuffle, he looks at the camera (and still turns to the arrow keys).
