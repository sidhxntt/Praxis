/**
 * LIMELIGHT · website hero
 *
 * The first screen of a brand studio whose pitch is being seen. On a yellow
 * ground printed as a halftone, a woman stands in front of one giant word as
 * a pure black silhouette. One soft, warm light travels slowly over her, and
 * wherever it falls the real colour photograph appears inside the black: an
 * eye, the other eye, a cheek, the mouth, the fringe, and back again. It is a
 * light, not a hole: brightest in its middle, falling off to black, shaped by
 * the face (a relief taken from the photo's own brightness), and everything
 * else answers it too. The halftone dots swell and glow under it, the letters
 * fill in white near it, a thin warm rim runs along the edge of her hair, and
 * her head throws a soft shadow onto the letters on the far side.
 *
 * The pointer is a torch: move over the picture and the light is in your
 * hand. Press anywhere for a studio flash: the whole portrait is lit for a
 * moment, then falls back to black (with a soft pop). Four planes (the print,
 * the word, the person, the copy and cards) part slowly in parallax on their
 * own and more under the pointer. The numbers count up once and hold.
 *
 * One file, React only. The typefaces come from Google Fonts (loaded in the CSS).
 *
 *   <Limelight />
 *   <Limelight word="NOTICED" tone="#fb8b1e" light={1.3} />
 *
 * Props
 *   brand     the wordmark                               "Limelight"
 *   word      the giant word behind her                  "SPOTTED"
 *   cta       the nav button                             "Start a project"
 *   tone      the ground colour                          "#f6b90c"
 *   light     the size of the light, 1 = normal          1
 *   portrait  the person, a transparent cut-out          "/design/websites/limelight/portrait.webp"
 *   faces     three square avatars side by side          "/design/websites/limelight/faces.webp"
 *   sound     a tiny synthesized pop on the flash        true
 *   onCta     called when the nav button is pressed
 *
 * Fills its container's width; height is var(--design-h, 100svh), never under
 * 640px. Responsive by container query: on a phone the word sits above her
 * head and the copy and cards stack over her jumper.
 * The scene pauses off screen and in a hidden tab. Without WebGL2 the same
 * composition is shown still (CSS halftone, the word, the silhouette with the
 * light resting on her eyes).
 * Reduced motion: no entrance, no count-up, one finished frame with the light
 * resting on her eyes.
 * Keyboard: the hidden "Move the light" button (first Tab stop after the nav)
 * moves the light with the arrow keys; Enter or Space fires the flash.
 * Sound: silent until the visitor interacts, when `sound` is false, and when
 * <html data-cd-mute="1">. The AudioContext is created on the first sound.

/* Sound: a tiny synthesized click or pop, made in the browser.
   Silent when `sound` is false, when <html data-cd-mute="1">, and until the
   visitor has interacted. Delete this hook and its calls to make it silent. */
type Blip = { f: number; to?: number; d?: number; g?: number; type?: OscillatorType; noise?: boolean; q?: number; at?: number };
function useBlip(on: boolean) {
  const ctx = useRef<AudioContext | null>(null);
  useEffect(() => () => void ctx.current?.close().catch(() => {}), []);
  return useCallback(
    (b: Blip) => {
      if (!on || typeof window === "undefined") return;
      if (document.documentElement.dataset.cdMute === "1") return;
      const ua = (navigator as unknown as { userActivation?: { hasBeenActive: boolean } }).userActivation;
      if (ua && !ua.hasBeenActive) return;
      try {
        const AC = window.AudioContext ?? (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
        if (!AC) return;
        const a = (ctx.current ??= new AC());
        if (a.state === "suspended") void a.resume();
        const t = a.currentTime + (b.at ?? 0);
        const d = Math.min(b.d ?? 0.035, 0.16);
        const amp = a.createGain();
        amp.gain.setValueAtTime(0.0001, t);
        amp.gain.exponentialRampToValueAtTime(Math.min(b.g ?? 0.035, 0.06), t + (b.noise ? 0.006 : 0.004));
        amp.gain.exponentialRampToValueAtTime(0.0001, t + d);
        amp.connect(a.destination);
        let src: AudioScheduledSourceNode;
        if (b.noise) {
          const buf = a.createBuffer(1, Math.ceil(a.sampleRate * d), a.sampleRate);
          const ch = buf.getChannelData(0);
          for (let i = 0; i < ch.length; i++) ch[i] = Math.random() * 2 - 1;
          const n = a.createBufferSource();
          n.buffer = buf;
          const bp = a.createBiquadFilter();
          bp.type = "bandpass";
          bp.frequency.setValueAtTime(b.f, t);
          if (b.to) bp.frequency.exponentialRampToValueAtTime(b.to, t + d);
          bp.Q.value = b.q ?? 1.2;
          n.connect(bp).connect(amp);
          src = n;
        } else {
          const o = a.createOscillator();
          o.type = b.type ?? "sine";
          o.frequency.setValueAtTime(b.f, t);
          if (b.to) o.frequency.exponentialRampToValueAtTime(b.to, t + d);
          o.connect(amp);
          src = o;
        }
        src.start(t);
        src.stop(t + d + 0.01);
      } catch {
        /* no sound is fine */
      }
    },
    [on],
  );
}

const hexRgb = (hex: string): [number, number, number] => {
  const h = hex.replace("#", "");
  const f = h.length === 3 ? h.split("").map((c) => c + c).join("") : h.padEnd(6, "0").slice(0, 6);
  const n = parseInt(f, 16);
  return Number.isFinite(n) ? [((n >> 16) & 255) / 255, ((n >> 8) & 255) / 255, (n & 255) / 255] : [0.965, 0.725, 0.047];
};

  [0.114, 0.468],
  [0.004, 0.562],

/* ---------- The lit picture (WebGL2) ----------
   One full-screen pass, three planes, all in layout px:
   ground  a 45° halftone screen; each dot's radius is the light at that dot:
           a warm halo behind her head, the uneven print, and the light itself.
   word    the letters from a canvas texture, fading out downward, filled in
           white where the light is near, with her head's shadow on them.
   person  black by default (a hair of the photo left in it, a backlit wrap at
           the edges); the light adds the colour photograph back, shaped by a
           relief from the photo's own blurred brightness. */
const VERT = `#version 300 es
in vec2 aPos;
void main(){ gl_Position = vec4(aPos, 0.0, 1.0); }`;

const FRAG = `#version 300 es
precision highp float;
uniform vec2 uRes;
uniform float uDpr, uIn, uFlash, uCell, uHL, uTL, uR, uTA, uBot;
uniform vec2 uL, uGO, uWY;
uniform vec4 uHR, uTR;
uniform vec3 uTone;
uniform sampler2D uHead, uText;
out vec4 o;

float bell(vec2 q, float s){ vec2 d = (q - uL) / s; return exp(-dot(d, d)); }
vec2 huv(vec2 q){ vec2 uv = (q - uHR.xy) / uHR.zw; return vec2(uv.x, min(uv.y, .998)); }
float headA(vec2 q, float lod){
  vec2 uv = (q - uHR.xy) / uHR.zw;
  if (uv.x < 0. || uv.x > 1. || uv.y < 0.) return 0.;
  return textureLod(uHead, vec2(uv.x, min(uv.y, .998)), lod).a;
}
float headL(vec2 q, float lod){ return dot(textureLod(uHead, huv(q), lod).rgb, vec3(.299, .587, .114)); }
float textA(vec2 q, float lod){
  vec2 uv = (q - uTR.xy) / uTR.zw;
  if (uv.x < 0. || uv.y < 0. || uv.x > 1. || uv.y > 1.) return 0.;
  return textureLod(uText, uv, lod).a;
}
float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
float vnoise(vec2 p){
  vec2 i = floor(p), f = fract(p); f = f * f * (3. - 2. * f);
  return mix(mix(hash(i), hash(i + vec2(1., 0.)), f.x), mix(hash(i + vec2(0., 1.)), hash(i + vec2(1., 1.)), f.x), f.y);
}

void main(){
  vec2 p = vec2(gl_FragCoord.x, uRes.y * uDpr - gl_FragCoord.y) / uDpr;
  float LI = uIn;

  // her head's shadow falls away from the light
  vec2 hc = uHR.xy + uHR.zw * vec2(.5, .327);
  vec2 dv = hc - uL;
  float dl = max(length(dv), 1.);
  vec2 dn = dv / dl;
  vec2 offT = dn * min(dl * .24, uHR.w * .075);
  float sT = headA(p - offT, uHL + 3.7);

  /* ground: a 45 degree screen */
  const float K = .70710678;
  vec2 g = p - uGO;
  vec2 gr = vec2((g.x - g.y) * K, (g.x + g.y) * K);
  vec2 cr = (floor(gr / uCell) + .5) * uCell;
  vec2 cc = vec2((cr.x + cr.y) * K, (cr.y - cr.x) * K) + uGO;
  vec2 hq = (cc - mix(hc, uL, .2)) / (uHR.w * vec2(.59, .53));
  float halo = exp(-dot(hq, hq));
  vec2 hq2 = (p - mix(hc, uL, .2)) / (uHR.w * vec2(.64, .57));
  float haloP = exp(-dot(hq2, hq2));
  float pch = vnoise(cc / 140. + 3.7) * .62 + vnoise(cc / 43. + 11.) * .38;
  float lg = bell(cc, uR * 2.5) * LI;
  float lgP = bell(p, uR * 2.7) * LI;
  float I = .02 + .22 * pch * pch + .74 * halo * halo + .16 * halo + .46 * lg + uFlash * .26;
  // grit: no two dots of a real print are the same size, and a few barely took
  float jt = hash(cr / uCell);
  float r = uCell * (.07 + .43 * sqrt(clamp(I, 0., 1.4))) * (.74 + .5 * jt) * mix(.6, 1., step(.03, jt));
  float aa = .75 / uDpr + .12;
  float m = 1. - smoothstep(r - aa, r + aa, length(gr - cr));
  vec3 inkA = mix(uTone * vec3(.98, .74, .6), vec3(.95, .34, .07), .42);
  vec3 inkB = mix(uTone * .9, vec3(.91, .16, .07), .78);
  vec3 ink = mix(inkA, inkB, smoothstep(.3, .95, I - lg * .25));
  ink = mix(ink, vec3(1., .47, .10), lg * .45);
  vec3 base = mix(uTone, vec3(.99, .52, .10), haloP * .50);
  base = mix(base, vec3(.95, .30, .07), haloP * haloP * haloP * .42);
  base = mix(base, vec3(1., .84, .34), lgP * .5);
  base += vec3(.05, .05, .02) * uFlash;
  vec3 col = mix(base, ink, m * mix(.62, .96, clamp(I, 0., 1.)));
  col += (hash(floor(p * uDpr / 1.5) + 7.) - .5) * .03;

  /* word */
  vec2 tuv = (p - uTR.xy) / uTR.zw;
  if (tuv.x > 0. && tuv.y > 0. && tuv.x < 1. && tuv.y < 1.) {
    float T = textureLod(uText, tuv, max(0., uTL - log2(uDpr))).a * uTA;
    if (T > .001) {
      float ty = clamp((p.y - uWY.x) / max(1., uWY.y - uWY.x), 0., 1.);
      float fade = mix(1., .14, smoothstep(.14, 1., ty));
      float Lw = bell(p, uR * 3.1) * LI;
      float lift = clamp(Lw * .9 + uFlash * .75, 0., 1.) * (1. - .85 * sT);
      fade = mix(fade, 1., lift);
      vec3 wc = mix(vec3(1., .975, .90), vec3(1.), clamp(Lw + uFlash, 0., 1.));
      col = mix(col, wc, T * fade);
      // her shadow on the letters
      col = mix(col, col * vec3(.70, .47, .25), T * sT * (.40 + .4 * bell(p, uR * 5.) * LI));
    }
  }

  // the floor of the picture goes a little darker, so the copy reads
  float by = smoothstep(uBot, 1., p.y / uRes.y);
  col = mix(col, col * vec3(.50, .40, .22), by * .62);

  /* person */
  vec2 uv = (p - uHR.xy) / uHR.zw;
  if (uv.x > 0. && uv.x < 1. && uv.y > 0.) {
    float l0 = max(0., uHL - log2(uDpr) - .35);
    vec4 h = textureLod(uHead, vec2(uv.x, min(uv.y, .998)), l0);
    if (h.a > .003) {
      vec3 c = h.rgb / h.a;
      // relief: brighter is nearer (nose, cheekbones, brow), darker falls back (eyes, lips, hair)
      float bl = uHL + 3.4;
      float e = uHR.w * .0082;
      float h0 = headL(p, bl), hx = headL(p + vec2(e, 0.), bl), hy = headL(p + vec2(0., e), bl);
      vec3 n = normalize(vec3((h0 - hx) * 3., (h0 - hy) * 3., 1.));
      vec3 l3 = normalize(vec3(uL - p, uR * .95));
      float ndl = clamp(dot(n, l3), 0., 1.);
      vec2 dd = (p - uL) / uR;
      float b = exp(-pow(dot(dd, dd), 1.2));
      float E = 1.5 * b * mix(.6, 1., ndl) * LI;
      E = max(E - .012, 0.) + uFlash * 1.06;
      vec3 warm = mix(vec3(1.07, .985, .87), vec3(1.03, 1., .96), clamp(uFlash * 1.4, 0., 1.));
      // the rows under the photograph's own frame are drawn in, not shot: they stay black
      vec3 lit = pow(c, vec3(1.05)) * warm * min(E, 1.32) * (1. - smoothstep(.845, .915, uv.y));
      // the black: a hair of the picture, and the ground's light wrapping round her edges
      float wrap = 1. - headA(p, uHL + 3.6);
      vec3 sil = c * .036 * vec3(1., .9, .78) + vec3(.55, .27, .05) * wrap * wrap * .34 * (.35 + c);
      // a thin warm rim on the edge that faces the light
      vec2 tl = uL - p;
      vec2 ld = tl / max(length(tl), 1.);
      float rim = clamp(1. - headA(p + ld * 3.2, l0 + .6), 0., 1.) + .35 * clamp(1. - headA(p + ld * 9., uHL + 1.8), 0., 1.);
      sil += vec3(1., .80, .42) * rim * bell(p, uR * 3.6) * LI * 1.25;
      col = mix(col, min(sil + lit, vec3(1.)), h.a);
    }
  }

  col += vec3(1., .96, .86) * uFlash * uFlash * .10;
  float nz = fract(sin(dot(floor(p * uDpr), vec2(12.9898, 78.233))) * 43758.5453);
  col += (nz - .5) * .02;
  o = vec4(col, 1.);
}`;

      x.fillStyle = "#fff";
      const txt = wi.textContent || "";
      const mt = x.measureText(txt);
      const A = mt.fontBoundingBoxAscent || fs * 0.88;
      const D = mt.fontBoundingBoxDescent || fs * 0.21;
      const base = pad + (lh - (A + D)) / 2 + A;
      x.fillText(txt, tw / 2, base);
      upload(1, textTex, tc);
      TR.x = (wr.left - r.left) / kScale - pad;
      TR.y = (wr.top - r.top) / kScale - pad;
      TR.w = tw;
      TR.h = th;
      TR.dpr = td;
      const cap = x.measureText("H").actualBoundingBoxAscent || fs * 0.69;

      TR.y1 = base;

      H = Math.max(1, el.offsetHeight);

      cv.height = Math.round(H * dpr);
      gl.viewport(0, 0, cv.width, cv.height);

      HR.y = (hr.top - r.top) / kScale;

      HR.h = Math.max(1, hr.height / kScale);
      headPx = imgH / HR.h;

    // draw, and draw again whenever a font finishes loading.

    const go = () => {

      fontsOk = true;
      layout();
      kick();

    const N = PATH.length;

    const path = (t: number): [number, number] => {
      const u = ((t / LAP) * N + 0.5) % N;
      const i = Math.floor(u);
      let s = u - i;
      s = s - (0.55 / (2 * Math.PI)) * Math.sin(2 * Math.PI * s);
      const P = (k: number) => PATH[(((i + k) % N) + N) % N];
      const [a, b, c, d] = [P(-1), P(0), P(1), P(2)];
      const cr = (j: 0 | 1) =>
        0.5 * (2 * b[j] + (c[j] - a[j]) * s + (2 * a[j] - 5 * b[j] + 4 * c[j] - d[j]) * s * s + (3 * b[j] - a[j] - 3 * c[j] + d[j]) * s * s * s);
      return at(cr(0), cr(1));

    const t0 = performance.now();
    let last = t0;
    let lx = 0;
    let ly = 0;
    let placed = false;
    let cx = 0;
    let cy = 0;
    let pcx = 0;
    let pcy = 0;
    let kf = 2.2; // how tightly the light follows its target
    let flashAt = -99;
    let liveSet = false;

    const frame = (now: number) => {
      raf = 0;
      if (dead) return;
      if (!visible || document.hidden) return;
      if (!still) raf = requestAnimationFrame(frame);

      last = now;
      const t = still ? 0 : (now - t0) / 1000;
      if (!headReady || !textReady) {

          ly = py;

        const k = 1 - Math.exp(-dt * kf);

        ly += (py - ly) * k;

        const ty = ptr.on && !ptr.key ? ((ptr.y / H) * 2 - 1) * 10 : 0;
        const kc = 1 - Math.exp(-dt * 2.6);
        pcx += (tx - pcx) * kc;
        pcy += (ty - pcy) * kc;

        cx = (Math.sin(t * 0.23) * 9 + pcx) * amp;
        cy = (Math.sin(t * 0.17 + 1) * 4.5 + pcy) * amp;
      }
      const intro = still ? 1 : 1 - Math.pow(1 - Math.min(1, Math.max(0, (t - 0.05) / 1.15)), 3);
      const fa = t - flashAt;
      const fl = still || fa < 0 || fa > 3 ? 0 : Math.min(1, fa / 0.05) * Math.exp(-Math.max(0, fa - 0.05) * 4.3);
      const rise = 1 - intro;
      const size = Math.min(2.2, Math.max(0.4, lightRef.current || 1));
      gl.useProgram(prog);
      gl.uniform2f(U.uRes, W, H);
      gl.uniform1f(U.uDpr, dpr);
      gl.uniform1f(U.uIn, intro);
      gl.uniform1f(U.uFlash, fl);
      gl.uniform1f(U.uCell, phone ? 6 : 7);
      gl.uniform1f(U.uTA, Math.min(1, intro * 1.8));
      gl.uniform1f(U.uBot, phone ? 0.52 : 0.84);
      gl.uniform1f(U.uR, HR.h * 0.127 * size * (still ? 1.2 : 1));
      gl.uniform1f(U.uHL, Math.log2(Math.max(1e-3, headPx)));
      gl.uniform1f(U.uTL, Math.log2(Math.max(1e-3, TR.dpr)));
      gl.uniform2f(U.uL, lx, ly);
      gl.uniform2f(U.uGO, -cx * 0.12, -cy * 0.12);
      gl.uniform4f(U.uHR, HR.x - cx, HR.y - cy, HR.w, HR.h);
      const wy = TR.y - cy * 0.42 + rise * 40;
      gl.uniform4f(U.uTR, TR.x - cx * 0.42, wy, TR.w, TR.h);
      gl.uniform2f(U.uWY, wy + TR.y0, wy + TR.y1);
      const [r, g, b] = hexRgb(toneRef.current);
      gl.uniform3f(U.uTone, r, g, b);
      gl.activeTexture(gl.TEXTURE0);
      gl.bindTexture(gl.TEXTURE_2D, headTex);
      gl.activeTexture(gl.TEXTURE1);
      gl.bindTexture(gl.TEXTURE_2D, textTex);
      gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
      if (copyRef.current) copyRef.current.style.transform = `translate3d(${(-cx * 1.4).toFixed(2)}px,${(-cy * 1.4 + rise * 18).toFixed(2)}px,0)`;
      if (cardsRef.current) cardsRef.current.style.transform = `translate3d(${(-cx * 1.7).toFixed(2)}px,${(-cy * 1.7 + rise * 26).toFixed(2)}px,0)`;
      if (!liveSet) {
        liveSet = true;
        setLive(true);
      }

# Limelight · brand studio hero

The first screen of a brand studio whose pitch is being seen ("We put brands where the light is"). On a saturated yellow ground printed as a halftone, a woman with a sharp black bob stands in front of one giant white word, SPOTTED, as a pure black silhouette. (The word is chosen so it still reads with her head over its middle: the hidden letter is one of the two Ts. NOTICED and VISIBLE are the other options.) One soft warm light travels slowly over her, and wherever it falls the real colour photograph appears inside the black: an eye, the other eye, a cheek, the mouth, the fringe, and back. It must read as light, not as a hole in a mask: brightest in its middle, falling off to black, slightly warm, shaped by the face. Everything else answers the same light: the halftone dots swell and glow under it, the letters fill in solid white near it, a thin warm rim runs along the edge of her hair that faces it, and her head throws a soft shadow onto the letters on the far side. The pointer is a torch in the visitor's hand; a press fires a studio flash that shows the whole portrait for a moment.

## Images (two)

1. `portrait.webp` (3046×2206, RGBA). Prompt (square, gpt-image class model):
   "Studio portrait photograph of an East Asian woman in her early thirties with a sharp, glossy, jet black blunt bob cut at jaw length with a heavy straight fringe just above the eyebrows, wearing a black fine-knit turtleneck. Evenly lit by one large soft key light from the front, slightly above, with gentle fill, so the whole face is clearly visible. Real skin texture with visible pores and tiny imperfections, natural unretouched look, sharp catchlights in the eyes. Calm, direct, confident gaze straight into the lens, mouth closed, head perfectly upright and symmetrical, facing the camera square on. Pure seamless bright white studio backdrop, evenly lit, no shadow on the backdrop, no vignette, no gradient. Editorial studio photograph shot on a medium format digital camera with a 110mm lens at f/8, everything in sharp focus including the hair edges. Framing: head and shoulders, the head centred and large, the whole hair inside the frame with a little white margin above it, both shoulders fully inside the frame with white margin left and right, the shoulders running out of the bottom edge. No jewelry, no text, no logo."
   Pick the person by two tests: the black silhouette alone is iconic at thumbnail size, and the lit face reads as a real studio photograph.
   Cut-out: because hair and jumper are black on white, key it by brightness instead of a subject mask. Background = the white region connected to the border; alpha in a 18 px band inside the edge = `(white − darkest channel) / (white − 105)` (so fine hairs keep partial alpha), fully opaque deeper in and on anything skin-coloured; un-mix the white out of the edge pixels and clamp them dark (no halo when the flash shows the hair). Small pockets of backdrop seen through the hair by her jaw are keyed the same way. Lanczos ×1.6, unsharp (radius 1.6, 60%). Then give the picture room: 200 more rows of jumper under the frame, and 520 px more on each side where every new column is the edge column slid down by `0.5k + 0.0009k²` px, so the shoulders keep sloping out of the bottom corners instead of ending in a vertical cut. cwebp q90, alpha q100, `-exact`.
   Where things are, in units of the picture's height, x from its centre line, y from its top: eyes (∓0.086, 0.365), cheeks (±0.115, 0.465), mouth (0, 0.562), fringe (0, 0.27), hair top 0.02, chin 0.61.
2. `faces.webp` (432×144): three square avatars side by side, cropped from one generated triptych of founder headshots on flat teal, coral and cobalt backdrops.

## Layers (back to front)

1. **Still fallback** (shown until WebGL2 draws, kept if it is missing): yellow `#f6b90c`; a CSS halftone (two offset radial-gradient dot layers, 1.1 px dots of `rgba(240,112,16,.8)` on a 10 px grid); an orange-red radial glow behind the head; the word as real text, white, masked from solid at 12% to 12% opacity at 88% of its height; the photo twice: once with `filter: brightness(0)` (the silhouette), once in colour under a radial mask centred on the eyes; a soft dark-olive gradient over the bottom 16%. The WebGL pass reads the boxes of the word and the photo from these DOM elements, so the layout is defined once, in CSS.
2. **WebGL2 canvas** (one context, DPR ≤ 2, one full-screen fragment pass, about 11 texture taps a pixel, paused off screen and in a hidden tab). All maths in layout px (when the hero is shown scaled down, the canvas resolution follows the scale).
   - The light `L`, radius `R = 0.127 × photo height × light` (about 98 px at 1440×810).
   - **Idle path**: a closed Catmull-Rom loop through seven points on her (left eye, right eye, right cheek, mouth, left cheek, left temple, fringe over the right eye), one lap 17.5 s. Within each leg the parameter is `s − (0.55/2π)·sin(2πs)`, so it slows to about 45% at each point and never stops. It starts between the eyes. It always stays on the face, so in any two seconds something of her is appearing or leaving.
   - **Pointer**: `L` eases to the pointer with `1 − e^(−k·dt)`, `k` gliding between 9 (in the hand) and 2.2 (returning to the path) at rate 3, so there is no overshoot and no jump when the pointer leaves.
   - **Ground**: a 45° screen, 7 px cells (6 px on phones). Light per cell `I = .02 + .22·n² + .74·halo² + .16·halo + .46·bell(L, 2.5R) + .26·flash`, where `n` is two octaves of value noise (140 px and 43 px, the unevenness of a print) and `halo` is a Gaussian behind her head (radii .59 × .53 of the photo height, so the heat shows well past the edges of her hair while the corners stay clean yellow, its centre pulled 20% toward the light). Dot radius `cell·(.07 + .43·√I)`, then the grit of a real print: each dot × `(.74 + .5·random)` and 3% of the dots at 60% size, plus ±1.5% grain on the ground; anti-aliased by 0.75 device px. Ink runs from orange (`tone × (.98,.74,.6)` mixed 42% with `#f25712`) to red (`tone × .9` mixed 78% with `#e82912`) as `I` rises, and toward `#ff781a` under the light. Under the dots the ground is the tone, mixed up to 50% with `#fc851a` across a slightly wider halo (radii .64 × .57) and a further 42% × halo³ with `#f24d12` at its core, and 50% with `#ffd657` in a pool under the light.
   - **Word**: Archivo 800, fitted in JS so it runs edge to edge (width = 101.2% of the hero, 20.3cqi for SPOTTED), line-height .8, centred at 58% of the height. Drawn white into a canvas texture with the page's font (waits for the font, redraws on `loadingdone`). Opacity runs from 1 at the cap top to .14 at the baseline (smoothstep from 14%); near the light it fills to solid (`+ .9·bell(L, 3.1R)`, less inside her shadow); colour `#fff9e6` to pure white near the light. Her shadow: the photo's alpha from a blurred mip (about 13 px), offset away from the light by `min(.24·d, 7.5% of the photo height)`, multiplies the letters by `(.70,.47,.25)` at 40 to 80%.
   - **Person**: black by default: 3.6% of the photo (warm tinted) plus a backlit wrap at the edges (`(1 − blurred alpha)² × (.55,.27,.05) × .34`). The light adds the photo back: `E = 1.5 · e^(−(d/R)^2.4) · mix(.6, 1, n·l)`, where `n` is a normal from the photo's own blurred brightness (brighter is nearer: gradient × 3 over 0.8% of the height) and `l` points to the light held `.95R` above the picture; lit colour `photo^1.05 × (1.07,.985,.87) × min(E, 1.32)`, faded to nothing between 84.5% and 91.5% of the picture's height, so the drawn-in rows under the photograph's own frame stay black even in the flash. Rim: `1 − alpha` sampled 3.2 px toward the light (plus 35% of a 9 px soft sample), × `bell(L, 3.6R)` × 1.25, colour `#ffcc6b`.
   - A dark-olive floor over the bottom 16% (48% on phones) so the copy reads; static grain ±1%.
   - **Flash** (press anywhere on the picture, or Enter/Space on the light button): `f = min(1, t/50 ms) · e^(−4.3·(t − 50 ms))`, about 0.9 s. It adds `1.06·f` to the exposure of the whole portrait (neutral white instead of warm), swells every dot, fills the word and lifts the frame by `.10·f²`. Sound: band-passed noise 3200→700 Hz for 110 ms, a 170→55 Hz thump for 90 ms and a 12 ms tick, all under 6% gain.
   - **Parallax**: camera `c = (sin(.23t)·9, sin(.17t+1)·4.5) px` plus the eased pointer (±18, ±10 px); × .45 on phones. Ground moves −.12c, word −.42c, person −1c, copy −1.4c, cards −1.7c.
   - **Entrance** (done by 1.2 s): the light fades up, the word rises 40 px, the copy 18 px, the cards 26 px; the canvas fades in over .4 s. The silhouette and ground are there from the first frame.
3. **UI** (Archivo for everything)

Dark card: a deep tint of the ground, `color-mix(in srgb, tone 27%, #170f03)` at 82% opacity (`rgba(72,54,10,.80)` on yellow), with a 14 px backdrop blur and a 1 px warm inner line: "Next opening" with a yellow dot, "March" 30/600, "2 slots left".

## Responsive

- ≤ 640 px (phone): the word moves above her (centred at 16.2% of the height, still edge to edge) with the top of her hair over its lower edge; the photo starts at 17.5% and is 66% of the height tall, so her head fills most of the width; the copy (17 px) and the two cards (side by side, flexible, 120 px tall) stack over her jumper, with the dark floor rising to 48%.

## Keyboard and reduced motion

A visually hidden button after the nav ("Move the light with the arrow keys, Enter fires the flash") becomes visible on focus: arrows move the light by 5% of the photo height, Enter or Space fires the flash, blur gives the light back to its path. Reduced motion: no entrance, no count-up, no path, no parallax: one finished frame with a slightly larger light resting between her eyes.
