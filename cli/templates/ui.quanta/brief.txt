"use client";

/**
 * QUANTA · website hero
 *
 * The launch screen of a fast reasoning model (or any inference API), and the
 * speed is the scene. A ring of light made of thousands of particles orbits
 * the product: you sit low beside it, so its far side is a razor line on the
 * horizon and its near side sweeps past the lens out of focus, in front of the
 * console. The console is live: real prompts are answered at the model's
 * speed (the whole answer lands in a blink, each token arriving hot and
 * cooling), while a second lane shows a typical model still writing. Every
 * run fires a bright bunch round the ring. A throughput reading rolls its
 * digits like an odometer. The pointer moves the camera, so the planes part.
 *
 * One file, React only. Space Grotesk + JetBrains Mono from Google Fonts (in the CSS).
 *
 *   <Quanta />
 *   <Quanta model="Quanta 3 Mini" tps={2400} tone="#0a7d4b" accent="#ffe14a" />
 *
 * Props
 *   brand    the wordmark                                   "Quanta"
 *   model    the model's name (console, lane, eyebrow)      "Quanta 3"
 *   tps      tokens per second it serves                    1240
 *   tone     the colour of the whole stage (any colour)     "#1230d0"
 *   accent   the live colour (dot, bunch, your lane)        "#d4ff3a"
 *   ask      which prompt runs first (0–3)                  0
 *   cta      the main button                                "Get an API key"
 *   onCta    called when the button is pressed
 *   sound    a soft click on a prompt, a short burst of ticks when the answer lands
 *            (only after a first click or key, silent while html[data-cd-mute="1"])   true
 *
 * Keyboard: the prompts are buttons (Enter or Space runs one). Fills its
 * container's width; height is var(--design-h, 100svh), never under 640px.
 * Responsive by container query. Paused off screen. Reduced motion: one still
 * frame of the ring, a finished answer, nothing loops.
 */

// A believable tokenizer for the demo: words, with long ones split in two.
function tokenize(text: string): string[] {
  const out: string[] = [];
  for (const w of text.match(/\S+\s*/g) ?? []) {
    const core = w.trimEnd();
    if (core.length > 7) {
      const cut = Math.ceil(core.length / 2);
      out.push(w.slice(0, cut), w.slice(cut));
    } else out.push(w);
  }
  return out;
}

const RIVAL_TPS = 92; // a typical frontier model, for the second lane
const FIRST = 180; // ms to first token
const RIVAL_FIRST = 620;
// Readings a stream might really show, as fractions of the nominal speed.
const READ = [0.991, 1.012, 0.984, 1.004, 1.021, 0.997, 0.978, 1.009, 1.016, 0.989, 1.002, 0.995, 1.024, 0.981, 1.007, 0.999, 1.013, 0.986, 1.003, 1.018, 0.992, 0.976, 1.01, 1.001, 0.988, 1.015, 0.996, 1.006];

type AudioWin = Window & { webkitAudioContext?: typeof AudioContext };

function Odometer({ value }: { value: number }) {
  const text = Math.round(value).toLocaleString("en-US");
  const chars = text.split("");
  return (
    <span className="qa-odo" role="img" aria-label={`${text} tokens per second`}>
      {chars.map((c, i) =>
        /\d/.test(c) ? (
          <span key={chars.length - i} className="qa-dig" aria-hidden>
            <span style={{ transform: `translateY(${-Number(c) * 10}%)` }}>
              {"0123456789".split("").map((d) => (
                <b key={d}>{d}</b>
              ))}
            </span>
          </span>
        ) : (
          <span key={`s${i}`} className="qa-sep" aria-hidden>
            {c}
          </span>
        ),
      )}
    </span>
  );
}

export default function Quanta({
  brand = "Quanta",
  model = "Quanta 3",
  tps = 1240,
  tone = "#1230d0",
  accent = "#d4ff3a",
  ask = 0,
  cta = "Get an API key",
  onCta,
  sound = true,
}: Props) {
  const root = useRef<HTMLDivElement>(null);
  const back = useRef<HTMLCanvasElement>(null);
  const front = useRef<HTMLCanvasElement>(null);
  const pivot = useRef<HTMLElement>(null);
  const audio = useRef<AudioContext | null>(null);
  const armed = useRef(false);
  const burstAt = useRef(0);
  const lastRun = useRef(0);
  const tpsRef = useRef(tps);
  const accentRef = useRef(accent);
  tpsRef.current = tps;
  accentRef.current = accent;

  const TOKS = useMemo(() => ASKS.map((x) => tokenize(x.a)), []);
  const first = Math.abs(ask) % ASKS.length;
  const [cur, setCur] = useState(first);
  const [runId, setRunId] = useState(0);
  const [shown, setShown] = useState(TOKS[first].length);
  const [rival, setRival] = useState(TOKS[first].length);
  const [tick, setTick] = useState(0);

  const toks = TOKS[cur];
  const n = toks.length;
  const mine = (FIRST + (n / tps) * 1000) / 1000;
  const theirs = (RIVAL_FIRST + (n / RIVAL_TPS) * 1000) / 1000;
  const landed = shown >= n;
  const raced = rival >= n;

  const play = useCallback(
    (kind: "click" | "land") => {
      if (!sound || !armed.current || document.documentElement.dataset.cdMute === "1") return;
      try {
        if (!audio.current) {
          const AC = window.AudioContext || (window as AudioWin).webkitAudioContext;
          if (!AC) return;
          audio.current = new AC();
        }
        const ac = audio.current;
        if (ac.state === "suspended") void ac.resume();
        const t = ac.currentTime;
        // "land" is the answer arriving: five ticks inside 55 ms, rising. "click" is one soft tap.
        const hits = kind === "land" ? [0, 0.011, 0.022, 0.033, 0.044] : [0];
        hits.forEach((d, i) => {
          const o = ac.createOscillator();
          const g = ac.createGain();
          o.type = kind === "land" ? "sine" : "triangle";
          o.frequency.setValueAtTime(kind === "land" ? 1900 + i * 240 : 620, t + d);
          g.gain.setValueAtTime(0.0001, t + d);
          g.gain.exponentialRampToValueAtTime(kind === "land" ? 0.03 : 0.05, t + d + 0.003);
          g.gain.exponentialRampToValueAtTime(0.0001, t + d + (kind === "land" ? 0.012 : 0.04));
          o.connect(g).connect(ac.destination);
          o.start(t + d);
          o.stop(t + d + 0.05);
        });
      } catch {
        /* no audio here */
      }
    },
    [sound],
  );

  const run = useCallback((i: number) => {
    lastRun.current = performance.now();
    burstAt.current = performance.now();
    setCur(i);
    setRunId((r) => r + 1);
  }, []);

  // One run: the answer streams at `tps`, the second lane at a typical model's pace.
  useEffect(() => {
    if (runId === 0) return;
    const count = TOKS[cur].length;
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
      setShown(count);
      setRival(count);
      return;
    }
    const dur = (count / tpsRef.current) * 1000;
    const rdur = (count / RIVAL_TPS) * 1000;
    const t0 = performance.now();
    let raf = 0;
    let done = false;
    setShown(0);
    setRival(0);
    const step = (now: number) => {
      const t = now - t0;
      const s = t < FIRST ? 0 : Math.min(count, Math.ceil(((t - FIRST) / dur) * count));
      const r = t < RIVAL_FIRST ? 0 : Math.min(count, Math.floor(((t - RIVAL_FIRST) / rdur) * count));
      setShown(s);
      setRival(r);
      if (s >= count && !done) {
        done = true;
        play("land");
      }
      if (r < count) raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
    return () => cancelAnimationFrame(raf);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [runId]);

  // The demo: a first run shortly after it appears, then the next prompt every few seconds.
  // (A member can remove this effect: the console then only runs when someone picks a prompt.)
  useEffect(() => {
    const el = root.current;
    if (!el || window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    let visible = true;
    const io = new IntersectionObserver(([e]) => (visible = e.isIntersecting));
    io.observe(el);
    let k = first;
    const start = window.setTimeout(() => run(k), 900);
    const id = window.setInterval(() => {
      if (!visible || document.hidden || performance.now() - lastRun.current < 6400) return;
      k = (k + 1) % ASKS.length;
      run(k);
    }, 800);
    const live = window.setInterval(() => visible && !document.hidden && setTick((x) => x + 1), 1100);
    return () => {
      io.disconnect();
      window.clearTimeout(start);
      window.clearInterval(id);
      window.clearInterval(live);
    };
  }, [first, run]);

  useEffect(() => () => void audio.current?.close(), []);

  // The ring: a real perspective projection of ~10,000 particles, drawn on two canvases
  // (the far half behind the console, the near half in front of it).
  useEffect(() => {
    const el = root.current;
    const cb = back.current;
    const cf = front.current;
    const pv = pivot.current;
    if (!el || !cb || !cf || !pv) return;
    const gb = cb.getContext("2d");
    const gf = cf.getContext("2d");
    if (!gb || !gf) return;
    const still = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    // Seeded, so the still frame is always the same picture.
    let seed = 20261118;
    const rnd = () => ((seed = (seed * 16807) % 2147483647) - 1) / 2147483646;
    const gauss = () => (rnd() + rnd() + rnd() + rnd() - 2) * 1.732;

    // radius, width, share of the particles, brightness
    const BANDS = [
      [1, 0.003, 0.36, 1],
      [1, 0.03, 0.18, 0.55],
      [0.905, 0.011, 0.12, 0.6],
      [1.09, 0.016, 0.11, 0.5],
      [0.79, 0.045, 0.08, 0.32],
      [2.5, 0.12, 0.15, 0.62], // a wide outer ring: you sit inside it, so it passes right by the lens
    ];
    const small = el.clientWidth < 640;
    const N = small ? 5200 : 10500;
    const BN = 640; // the bunch fired by a run
    const ang = new Float32Array(N + BN);
    const rad = new Float32Array(N + BN);
    const hgt = new Float32Array(N + BN);
    const spd = new Float32Array(N);
    const siz = new Float32Array(N + BN);
    const bri = new Float32Array(N);
    const hotp = new Float32Array(N); // a few particles catch the light: they are the bokeh you notice
    let c = 0;
    BANDS.forEach(([r, w, share, b], k) => {
      const count = k === BANDS.length - 1 ? N - c : Math.round(N * share);
      for (let i = 0; i < count && c < N; i++, c++) {
        const rr = r + gauss() * w;
        ang[c] = rnd() * Math.PI * 2;
        rad[c] = rr;
        hgt[c] = gauss() * w * 0.45;
        spd[c] = Math.pow(rr, -1.5) * (1 + gauss() * 0.03);
        siz[c] = 0.7 + Math.pow(rnd(), 3) * 1.5;
        bri[c] = b * (0.45 + rnd() * 0.55);
        hotp[c] = rnd() < 0.07 ? 3.5 + rnd() * 4 : 0.75;
      }
    });
    for (let j = 0; j < BN; j++) {
      // offsets behind the head of the bunch: a dense head and a long tail
      ang[N + j] = -Math.abs(gauss()) * 0.035 - Math.pow(rnd(), 2.2) * 0.9;
      rad[N + j] = 1 + gauss() * 0.006;
      hgt[N + j] = gauss() * 0.003;
      siz[N + j] = 0.9 + rnd() * 1.2;
    }
    // Far dust, well beyond the ring.
    const SN = 420;
    const dust = new Float32Array(SN * 4);
    for (let i = 0; i < SN; i++) {
      dust[i * 4] = Math.PI * (0.12 + rnd() * 0.76);
      dust[i * 4 + 1] = 6 + rnd() * 7;
      dust[i * 4 + 2] = -0.4 + Math.pow(rnd(), 1.4) * 5.6;
      dust[i * 4 + 3] = 0.12 + Math.pow(rnd(), 3) * 0.6;
    }

    const disc = (color: string) => {
      const s = document.createElement("canvas");
      s.width = s.height = 64;
      const g = s.getContext("2d");
      if (g) {
        const grad = g.createRadialGradient(32, 32, 0, 32, 32, 32);
        grad.addColorStop(0, color);
        grad.addColorStop(0.42, color);
        grad.addColorStop(1, "rgba(255,255,255,0)");
        g.globalAlpha = 0.95;
        g.fillStyle = grad;
        g.fillRect(0, 0, 64, 64);
      }
      return s;
    };
    const white = disc("#ffffff");
    let hotColor = accentRef.current;
    let hot = disc(hotColor);

    let W = 0;
    let H = 0;
    let dpr = 1;
    let cx = 0;
    let cy = 0;
    let f = 1000;
    let h0 = 0.31;
    const D = 2.25;
    const fit = () => {
      W = el.clientWidth;
      H = el.clientHeight;
      dpr = Math.min(2, window.devicePixelRatio || 1);
      if (W * H * dpr * dpr > 3.4e6) dpr = Math.max(1, Math.sqrt(3.4e6 / (W * H)));
      for (const cv of [cb, cf]) {
        cv.width = Math.round(W * dpr);
        cv.height = Math.round(H * dpr);
      }
      cx = pv.offsetLeft;
      cy = pv.offsetTop;
      f = pv.offsetWidth * 2.27; // the pivot's width is the ring's half width on screen
      h0 = Math.max(0.1, pv.offsetHeight / 100); // and its height is how high the camera sits (31px = 0.31 radii)
      if (still || !visible) draw(performance.now());
    };

    // camera
    let tx = 0;
    let ty = 0;
    let px = 0;
    let py = 0;
    let ca = 1;
    let sa = 0;
    let cp = 1;
    let sp = 0;
    let h = 0.31;
    let ox = 0;
    let oy = 0;
    let oz = 0;
    let ol = 0;
    const proj = (a: number, r: number, y: number) => {
      const x = r * Math.cos(a);
      const z = r * Math.sin(a);
      const x1 = x * ca - z * sa;
      const z1 = x * sa + z * ca;
      const Y = y - h;
      const Z = z1 + D;
      const Z2 = Z * cp - Y * sp;
      oz = Z2;
      ol = z1 > 0 ? 0 : 1;
      if (Z2 < 0.1) return false;
      ox = cx + (f * x1) / Z2;
      oy = cy - (f * (Y * cp + Z * sp)) / Z2;
      return true;
    };

    let visible = true;
    let raf = 0;
    let last = performance.now();
    const born = last;
    let bAng = -Math.PI / 2 - 0.2; // where the bunch rests in the still frame: just left of centre, in front
    let bVel = 4;
    let bSeen = 0;
    const LEVELS = [0.34, 0.6, 0.95];
    const WIDTHS = [1, 1.25, 1.7];

    const ringGlow = (R: number, alpha: number, width: number) => {
      const SEG = 150;
      let lx = 0;
      let ly = 0;
      let ok = false;
      for (let i = 0; i <= SEG; i++) {
        const a = (i / SEG) * Math.PI * 2;
        const vis = proj(a, R, 0);
        if (vis && ok) {
          const g = ol ? gf : gb;
          const coc = 0.009 * f * Math.abs(1 / oz - 0.345);
          g.lineWidth = Math.max(1, (width * f) / oz + coc * 2);
          g.strokeStyle = `rgba(255,255,255,${(alpha / (1 + coc * 0.5)).toFixed(4)})`;
          g.beginPath();
          g.moveTo(lx, ly);
          g.lineTo(ox, oy);
          g.stroke();
        }
        lx = ox;
        ly = oy;
        ok = vis;
      }
    };

    const draw = (now: number) => {
      const dt = still ? 0.016 : Math.min(0.05, (now - last) / 1000);
      last = now;
      px += (tx - px) * 0.06;
      py += (ty - py) * 0.06;
      el.style.setProperty("--qa-px", px.toFixed(3));
      el.style.setProperty("--qa-py", py.toFixed(3));
      const az = px * 0.075;
      h = h0 - py * 0.05;
      const pitch = Math.atan2(h, D);
      cp = Math.cos(pitch);
      sp = Math.sin(pitch);
      ca = Math.cos(az);
      sa = Math.sin(az);

      const since = burstAt.current ? (now - burstAt.current) / 1000 : 99;
      const spin = still ? 1 : 1 + 2.4 * Math.exp(-(now - born) / 900); // it spins up fast, then settles
      const w0 = 0.34 * Math.pow(Math.max(200, tpsRef.current) / 1240, 0.6) * (1 + 1.7 * Math.exp(-since / 0.5)) * spin;
      const fade = still ? 1 : Math.min(1, (now - born) / 1300);
      const shutter = 0.024;

      for (const g of [gb, gf]) {
        g.setTransform(dpr, 0, 0, dpr, 0, 0);
        g.clearRect(0, 0, W, H);
        g.globalCompositeOperation = "lighter";
        g.lineCap = "round";
        g.globalAlpha = 1;
      }

      // far plane: dust and one distant ring
      gb.fillStyle = "#fff";
      for (let i = 0; i < SN; i++) {
        if (!proj(dust[i * 4], dust[i * 4 + 1], dust[i * 4 + 2])) continue;
        if (ox < 0 || ox > W || oy < 0 || oy > H) continue;
        gb.globalAlpha = dust[i * 4 + 3] * fade * 0.5;
        gb.fillRect(ox, oy, 1.3, 1.3);
      }
      gb.globalAlpha = fade;
      gf.globalAlpha = fade;
      if (h0 < 0.5) ringGlow(4.4, 0.14, 0.0018);
      // the bloom of the main ring
      ringGlow(1, 0.045, 0.05);
      ringGlow(1, 0.1, 0.014);
      ringGlow(1, 0.34, 0.0012);

      const bp = [new Path2D(), new Path2D(), new Path2D()];
      const fp = [new Path2D(), new Path2D(), new Path2D()];
      for (let i = 0; i < N; i++) {
        const w = spd[i] * w0;
        let a = ang[i] + w * dt;
        if (a > 6.2831853) a -= 6.2831853;
        ang[i] = a;
        const r = rad[i];
        if (!proj(a, r, hgt[i])) continue;
        const sx = ox;
        const sy = oy;
        if (sx < -90 || sx > W + 90 || sy < -90 || sy > H + 90) continue;
        if (r > 2 && !ol) continue; // the outer ring is only for the lens: its far side would cross the words
        const coc = 0.009 * f * Math.abs(1 / oz - 0.345);
        const size = siz[i] * 0.5 + coc;
        const g = ol ? gf : gb;
        if (size < 1.9) {
          // in focus: a fine streak, as long as the shutter is open
          const k = bri[i] > 0.7 ? 2 : bri[i] > 0.42 ? 1 : 0;
          const path = ol ? fp[k] : bp[k];
          proj(a - w * shutter, r, hgt[i]);
          path.moveTo(ox, oy);
          path.lineTo(sx + 0.01, sy);
        } else {
          // out of focus: a soft disc, stretched along its path by its own speed
          proj(a - w * shutter * 1.6, r, hgt[i]);
          const dx = sx - ox;
          const dy = sy - oy;
          const len = Math.sqrt(dx * dx + dy * dy) || 0.001;
          const d = Math.min(size, 90) * 2;
          const L = (len + d) / 64;
          const T = d / 64;
          const ux = dx / len;
          const uy = dy / len;
          g.globalAlpha = Math.min(1, bri[i] * hotp[i] * (1.5 / (1 + (size * size) / 3.2)) + 0.006) * fade;
          g.setTransform(dpr * ux * L, dpr * uy * L, -dpr * uy * T, dpr * ux * T, dpr * (sx + ox) * 0.5, dpr * (sy + oy) * 0.5);
          g.drawImage(white, -32, -32);
        }
      }
      gb.setTransform(dpr, 0, 0, dpr, 0, 0);
      gf.setTransform(dpr, 0, 0, dpr, 0, 0);
      for (let k = 0; k < 3; k++) {
        for (const [g, p] of [
          [gb, bp[k]],
          [gf, fp[k]],
        ] as const) {
          g.globalAlpha = fade;
          g.lineWidth = WIDTHS[k];
          g.strokeStyle = `rgba(255,255,255,${LEVELS[k]})`;
          g.stroke(p);
        }
      }

      // the bunch: fired by a run, it whips round the near side and fades over two seconds
      if (burstAt.current && burstAt.current !== bSeen) {
        bSeen = burstAt.current;
        bAng = Math.PI - 0.42;
        bVel = 10.5;
      }
      const bAlpha = still ? 0.9 : Math.min(1, Math.max(0, 1.25 - since / 3.2));
      if (bAlpha > 0.01) {
        if (!still) {
          bVel += (w0 - bVel) * (1 - Math.exp(-dt / 0.7));
          bAng += bVel * dt;
        }
        if (accentRef.current !== hotColor) {
          hotColor = accentRef.current;
          hot = disc(hotColor);
        }
        const hb = new Path2D();
        const hf = new Path2D();
        for (let j = 0; j < BN; j++) {
          const a = bAng + ang[N + j] * (0.35 + Math.min(1, bVel / 6));
          if (!proj(a, rad[N + j], hgt[N + j])) continue;
          const sx = ox;
          const sy = oy;
          if (sx < -70 || sx > W + 70 || sy < -70 || sy > H + 70) continue;
          const coc = 0.009 * f * Math.abs(1 / oz - 0.345);
          const size = siz[N + j] * 0.5 + coc;
          if (size < 2.2) {
            const path = ol ? hf : hb;
            proj(a - bVel * shutter, rad[N + j], hgt[N + j]);
            path.moveTo(ox, oy);
            path.lineTo(sx + 0.01, sy);
          } else {

            const d = Math.min(size, 60) * 2;
            g.globalAlpha = Math.min(1, 2.6 / (1 + (size * size) / 6) + 0.04) * bAlpha;
            g.drawImage(hot, sx - d / 2, sy - d / 2, d, d);
          }
        }
        for (const [g, p] of [
          [gb, hb],
          [gf, hf],
        ] as const) {
          g.globalAlpha = bAlpha;
          g.lineWidth = 1.9;
          g.strokeStyle = hotColor;
          g.stroke(p);
        }
        // its comet: a line of light along the ring behind the head, and the head's own glow
        const tail = 0.16 + Math.min(1, bVel / 6) * 0.75;
        let lx = 0;
        let ly = 0;
        let ok = false;
        for (let k = 0; k <= 30; k++) {
          const vis = proj(bAng - (k / 30) * tail, 1, 0);
          if (vis && ok) {
            const g = ol ? gf : gb;
            const coc = 0.009 * f * Math.abs(1 / oz - 0.345);
            const fall = (1 - k / 30) * (1 - k / 30);
            g.strokeStyle = hotColor;
            g.lineWidth = (0.007 * f) / oz + coc * 1.5;
            g.globalAlpha = (bAlpha * fall * 0.55) / (1 + coc * 0.25);
            g.beginPath();
            g.moveTo(lx, ly);
            g.lineTo(ox, oy);
            g.stroke();
            g.lineWidth = (0.03 * f) / oz + coc * 2.5;
            g.globalAlpha = (bAlpha * fall * 0.12) / (1 + coc * 0.2);
            g.stroke();
          }
          lx = ox;
          ly = oy;
          ok = vis;
        }
        if (proj(bAng, 1, 0)) {
          const g = ol ? gf : gb;
          const d = Math.min(320, (0.14 * f) / oz);
          g.globalAlpha = 0.55 * bAlpha;
          g.drawImage(hot, ox - d / 2, oy - d / 2, d, d);
          g.globalAlpha = 0.9 * bAlpha;
          g.drawImage(white, ox - d / 8, oy - d / 8, d / 4, d / 4);
        }

    if (still) draw(performance.now());

      ty = 0;

      el.addEventListener("pointerleave", leave);

The full brief of this design, for reference, or to rebuild it in another stack:

# Quanta: fast reasoning model / inference API launch hero

Build the first screen of an AI lab launching a fast reasoning model (it works for any inference API). The speed is the scene: a ring of light made of thousands of particles orbits the live console. You sit low beside the ring, so its far side is a razor line on the horizon and its near side sweeps past the lens out of focus, in front of the console. The idea you feel: **press a prompt, the whole answer lands in a blink, and a bright bunch whips round the ring.** No pictures: everything is drawn in code.

## Stage (one colour, lit from the horizon)
`--tone: #1230d0` (cobalt) is a prop; every stage colour is mixed from it, so any brand colour works.
- Ground: `color-mix(in oklab, tone, #000 78%)`.
- Sky layer (4% larger than the hero, shifts with the pointer by 1.2% / 0.8%): a soft light where the ring meets the horizon, `radial-gradient(54% 17% at 50% 60%, mix(tone, #fff 38%), transparent)`, over `radial-gradient(92% 74% at 50% 66%, tone 0%, mix(tone, #000 22%) 40%, mix(tone, #000 56%) 74%, ground 100%)`.
- Film grain on top of everything (SVG fractal noise, overlay, 20%).

## The ring (two canvases: the far half behind the console, the near half in front of it)
A real perspective projection, not a drawn ellipse. World: a ring of radius 1 lying flat. Camera 2.25 radii from the centre and 0.31 radii above the plane, pitched down to look at the centre. Focal length = (the ring's half-width on screen) × 2.27; the half-width is 56% of the hero's width on desktop, so the ring is wider than the screen. The ring's centre sits at 64.5% of the hero's height. Points with z > 0 go to the back canvas, the rest to the front canvas (z-index above the console, `pointer-events: none`). Both use additive blending.

- **10,500 particles** (5,200 on a phone), seeded so the still frame never changes, in bands of (radius, width, share, brightness): (1.0, .003, 36%, 1) the razor core · (1.0, .03, 18%, .55) its halo · (.905, .011, 12%, .6) · (1.09, .016, 11%, .5) · (.79, .045, 8%, .32) · and a wide outer ring (2.5, .12, 15%, .62) that you sit inside: only its near half is drawn, so it passes right by the lens as giant soft discs.
- Each orbits at `0.34 rad/s × (tps/1240)^0.6 × radius^-1.5`. So the near side crosses the screen at about 500 px/s and the far side at 170.
- **Depth of field**: circle of confusion `0.009 × f × |1/z − 0.345|` px. In focus (under 1.9 px) a particle is a fine streak from where it was 24 ms ago to where it is (three brightness levels, widths 1 / 1.25 / 1.7 px). Out of focus it is a soft white disc stretched along its own path (a 64 px radial sprite drawn through a rotated, scaled transform), alpha `brightness × 1.5 / (1 + size²/3.2)`; 7% of the particles are 3.5–7.5× brighter, and those are the bokeh you notice.
- **Lines of light**: the ring itself is also stroked in 150 segments whose width grows as it nears the lens: a wide bloom (5% white), a narrower one (10%) and a thin core (34%). One more faint ring at radius 4.4 draws a horizon line behind the console. 420 dust points far beyond (radius 6–13) fill the sky.
- **Pointer**: the camera orbits ±4.3° and rises or sinks 0.05 radii, eased at 0.06 per frame, so the dust, the far side and the near side slide against each other.
- **A run**: the ring surges (speed ×2.7, decaying with τ 0.5 s) and a bunch of 640 accent-coloured particles is fired from the left side at 10.5 rad/s, slowing toward the ring's speed (τ 0.7 s) and fading over 4 s: a dense head, a long tail, a line of accent light along the ring behind the head and a glow on the head. It whips across the front of the console as the answer lands, goes round the back, and comes back slowly.
- On load the ring spins up 3.4× fast and settles over a second while it fades in.

## The console's behaviour
- Answers are split into tokens (words; words over 7 letters in two). A run: 180 ms to first token, then tokens appear at `tps` (61 tokens in about 50 ms: three or four frames). Each token lands in the accent colour with a glow and cools to white over 750 ms, so the arrival is visible. When the last one lands the console's border flashes the accent for 900 ms.
- The second lane is a typical model: first token after 620 ms, then 92 tokens a second, counting "34 of 61 tokens" until it is done. Then the header says how many times sooner (total time ÷ total time).
- It runs once 0.9 s after it appears, then the next prompt every ~7 s (not while off screen, and not within 6.4 s of a click). Clicking a chip runs that prompt at once.
- The reading: `tps ×` a fixed list of 28 factors between 0.976 and 1.024, one every 1.1 s. Each digit is a column of 0–9 that slides to its value (700 ms, `cubic-bezier(.3,1.2,.4,1)`).
- Prompts and answers (real, correct): the proof that there are infinitely many primes; a reply to a customer who was charged twice (order 4417, $48.00 refunded, $10 credit); two-sum in O(n) with a hash map; why the sky is blue and sunsets red.
